------------------------------------------------------------ helpers
local function new(cls, props, kids)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(kids or {}) do c.Parent = o end
    return o
end
local function tw(o, t, p, style, dir)
    local tn = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), p)
    tn:Play()
    return tn
end
local function corner(r) return new("UICorner", {CornerRadius = UDim.new(0, r)}) end
local function stroke(col, th, tr)
    return new("UIStroke", {Color = col or C.PURPLE, Thickness = th or 1, Transparency = tr or 0.75, ApplyStrokeMode = Enum.ApplyStrokeMode.Border})
end
local function pad(n)
    return new("UIPadding", {PaddingTop = UDim.new(0, n), PaddingBottom = UDim.new(0, n), PaddingLeft = UDim.new(0, n), PaddingRight = UDim.new(0, n)})
end
local function grad(a, b, rot)
    return new("UIGradient", {Rotation = rot or 0, Color = ColorSequence.new({CK(0, a), CK(1, b)})})
end
local function tri(a, b, c, rot)
    return new("UIGradient", {Rotation = rot or 0, Color = ColorSequence.new({CK(0, a), CK(0.5, b), CK(1, c)})})
end
local function hover(b, over, out)
    b.MouseEnter:Connect(function() tw(b, 0.15, over) end)
    b.MouseLeave:Connect(function() tw(b, 0.2, out) end)
end
local function press(b)
    local s = new("UIScale", {Parent = b})
    b.MouseButton1Down:Connect(function() tw(s, 0.08, {Scale = 0.95}) end)
    local function up() tw(s, 0.25, {Scale = 1}, Enum.EasingStyle.Back) end
    b.MouseButton1Up:Connect(up)
    b.MouseLeave:Connect(up)
end
local function popIn(o, delay)
    local s = o:FindFirstChild("PopScale") or new("UIScale", {Name = "PopScale", Parent = o})
    s.Scale = 0.94
    task.delay(delay or 0, function()
        if s.Parent then tw(s, 0.4, {Scale = 1}, Enum.EasingStyle.Back) end
    end)
end
local function label(parent, props)
    local d = {BackgroundTransparency = 1, Font = Enum.Font.Gotham, TextSize = 14, TextColor3 = C.TEXT, TextXAlignment = Enum.TextXAlignment.Left, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    return new("TextLabel", d)
end
local function button(parent, props, kids)
    local d = {AutoButtonColor = false, Text = "", BorderSizePixel = 0, Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = C.TEXT, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    local b = new("TextButton", d, kids)
    press(b)
    return b
end
local function glass(parent, props, r)
    local d = {BackgroundColor3 = C.PANEL2, BackgroundTransparency = 0.35, BorderSizePixel = 0, Parent = parent}
    for k, v in pairs(props or {}) do d[k] = v end
    local f = new("Frame", d)
    corner(r or 14).Parent = f
    stroke(C.PURPLE, 1, 0.78).Parent = f
    new("UIGradient", {Rotation = 90, Color = ColorSequence.new({CK(0, WHITE), CK(1, Color3.fromRGB(185, 165, 210))}), Parent = f})
    return f
end
local function bar(parent, w, h, x, y, rot, col)
    return new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(w, h), Position = UDim2.new(0.5, x, 0.5, y), Rotation = rot or 0,

local function copyText(s)
    local fn = (typeof(setclipboard) == "function" and setclipboard)
        or (typeof(toclipboard) == "function" and toclipboard)
        or (typeof(Clipboard) == "table" and Clipboard.set)
    if not fn then return false end
    return (pcall(fn, s))
end
local function fmtTime(s)
    s = math.max(0, math.floor(s))
    return string.format("%02d:%02d:%02d", math.floor(s / 3600), math.floor(s / 60) % 60, s % 60)
end
local function nowStr(fmt)
    local ok, s = pcall(os.date, fmt)
    return ok and s or "--"
end

local function makeDraggable(handle, target)
    local dragging, start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, start, startPos = true, i.Position, target.Position
            target:SetAttribute("moved", false)
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - start
            if d.Magnitude > 5 then target:SetAttribute("moved", true) end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function navIcon(parent, kind)
    local box = new("Frame", {Position = UDim2.fromOffset(10, 8), Size = UDim2.fromOffset(14, 14), BackgroundTransparency = 1, ZIndex = 4, Parent = parent})
    local parts, rings = {}, {}
    local function sq(x, y, w, h, r)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h), BackgroundColor3 = C.MUTED, BorderSizePixel = 0, ZIndex = 4, Parent = box}, {corner(r or 2)})
        table.insert(parts, f)
        return f
    end
    local function ringF(x, y, d, th)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(d, d), BackgroundTransparency = 1, ZIndex = 4, Parent = box}, {new("UICorner", {CornerRadius = UDim.new(0.5, 0)})})
        local st = new("UIStroke", {Color = C.MUTED, Thickness = th, Parent = f})
        table.insert(rings, st)
        return f
    end
    if kind == "Home" then
        sq(2, 6, 10, 8, 2)
        local rf = sq(2.5, 1, 9, 9, 1)
        rf.Rotation = 45
    elseif kind == "Me" then
        ringF(4, 0, 6, 2)
        sq(1, 8, 12, 6, 3)
    elseif kind == "Utility" then
        sq(0, 0, 6, 6, 2) sq(8, 0, 6, 6, 2) sq(0, 8, 6, 6, 2) sq(8, 8, 6, 6, 2)
    elseif kind == "Auto" then
        ringF(1, 1, 12, 2)
        sq(5, 5, 4, 4, 2)
    elseif kind == "ESP" then
        sq(0, 5, 14, 4, 2)
        ringF(4, 3, 6, 2)
    elseif kind == "Move" then
        ringF(4, 0, 6, 2)
        sq(6, 6, 2, 8, 1)
        sq(1, 7, 12, 2, 1)
    elseif kind == "Farm" then
        sq(6, 5, 2, 9, 1)
        sq(0, 1, 6, 5, 3)
        sq(8, 0, 6, 5, 3)
    elseif kind == "Shop" then
        sq(1, 5, 12, 9, 2)
        ringF(3, 0, 8, 2)
    else -- Settings (gear)
        ringF(3, 3, 8, 2)
        sq(6, 0, 2, 3, 1) sq(6, 11, 2, 3, 1) sq(0, 6, 3, 2, 1) sq(11, 6, 3, 2, 1)
        local t1 = sq(1.6, 1.6, 2, 3, 1) t1.Rotation = -45
        local t2 = sq(10.4, 1.6, 2, 3, 1) t2.Rotation = 45
        local t3 = sq(1.6, 9.4, 2, 3, 1) t3.Rotation = 45
        local t4 = sq(10.4, 9.4, 2, 3, 1) t4.Rotation = -45
    end
    return function(col)
        for _, p in ipairs(parts) do p.BackgroundColor3 = col end
        for _, r in ipairs(rings) do r.Color = col end
    end
end

local function setPage(n)
    current = n
    for k, p in pairs(pages) do p.Visible = (k == n) end
    if S.anim then
        local i = 0
        for _, c in ipairs(pages[n]:GetChildren()) do
            if c:IsA("GuiObject") then
                i = i + 1
                popIn(c, (i - 1) * 0.045)
            end
        end
    end
    for k, b in pairs(navBtns) do
        local on = (k == n)
        tw(b, 0.2, {BackgroundTransparency = on and 0.7 or 1})
        tw(navStroke[k], 0.2, {Transparency = on and 0.6 or 1})
        tw(navTxt[k], 0.2, {TextColor3 = on and C.TEXT or C.MUTED})
        tw(navInd[k], 0.2, {BackgroundTransparency = on and 0 or 1})
        navIco[k](on and C.RED or C.MUTED)
    end
    titleLbl.Text = T(PAGEKEY[n])
end

local function slider(list, key, mn, mx, def, lo, onChange, step, suffix)
    local f = row(list, lo, 58)
    tl(label(f, {Position = UDim2.fromOffset(14, 6), Size = UDim2.new(1, -100, 0, 20), Font = Enum.Font.GothamMedium, TextSize = 13}), key)
    local val = label(f, {AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -14, 0, 6), Size = UDim2.fromOffset(70, 20), Text = tostring(def),
        TextXAlignment = Enum.TextXAlignment.Right, Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = C.RED})
    local track = new("Frame", {Position = UDim2.new(0, 14, 0, 40), Size = UDim2.new(1, -28, 0, 6), BackgroundColor3 = C.OFF, BorderSizePixel = 0, Parent = f}, {corner(3)})
    local fill = new("Frame", {Size = UDim2.fromScale(0, 1), BackgroundColor3 = WHITE, BorderSizePixel = 0, Parent = track}, {corner(3), tri(C.RED, C.PURPLE, C.BLUE, 0)})
    local knob = new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0, 0.5), Size = UDim2.fromOffset(14, 14), BackgroundColor3 = WHITE, BorderSizePixel = 0, ZIndex = 3, Parent = track},
        {corner(7), stroke(C.PURPLE, 2, 0.1)})
    local hit = new("TextButton", {Text = "", BackgroundTransparency = 1, Position = UDim2.new(0, -8, 0.5, -14), Size = UDim2.new(1, 16, 0, 28), Parent = track})
    local function setRatio(r, fire)
        r = math.clamp(r, 0, 1)
        local v = mn + (mx - mn) * r
        if step then v = math.floor(v / step + 0.5) * step else v = math.floor(v + 0.5) end
        v = tonumber(string.format("%.2f", math.clamp(v, mn, mx)))
        fill.Size = UDim2.fromScale(r, 1)
        knob.Position = UDim2.fromScale(r, 0.5)
        val.Text = tostring(v) .. (suffix or "")
        if fire and onChange then onChange(v) end
    end
    setRatio((def - mn) / (mx - mn), false)
    local dragging = false
    local function fromInput(i)
        setRatio((i.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), true)
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            list.ScrollingEnabled = false
            fromInput(i)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then fromInput(i) end
    end)
    bind(UIS.InputEnded, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
            dragging = false
            list.ScrollingEnabled = true
        end
    end)
end

            Lighting.FogEnd = 100000
            Lighting.Ambient = Color3.fromRGB(178, 178, 178)

            Lighting.Ambient = ORIG.Ambient
            Lighting.OutdoorAmbient = ORIG.OutdoorAmbient

    pcall(function()
        if S.boost then
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        elseif ORIGQ then

        end
    end)

        end
    end)
end

local function espClear(model)
    local o = EObjs[model]
    if o then
        if o.hl then o.hl:Destroy() end
        if o.bb then o.bb:Destroy() end
        EObjs[model] = nil
    end
end
local function espBuild(model, name, isNpc)
    espClear(model)
    local head = model:FindFirstChild("Head") or model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
    if not head then return end
    local hl = new("Highlight", {Name = "NX_ESP", Adornee = model, FillColor = C.RED, OutlineColor = WHITE, FillTransparency = 0.6,
        OutlineTransparency = 0, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop, Parent = model})
    local bb = new("BillboardGui", {Name = "NX_TAG", Adornee = head, AlwaysOnTop = true, Size = UDim2.fromOffset(190, 24),
        StudsOffset = Vector3.new(0, 2.8, 0), Parent = head})
    local lbl = new("TextLabel", {BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1), Font = Enum.Font.GothamBold, TextSize = 13,
        TextColor3 = WHITE, TextStrokeTransparency = 0.5, Text = name, Parent = bb})
    EObjs[model] = {hl = hl, bb = bb, lbl = lbl, head = head, name = name, isNpc = isNpc, hum = model:FindFirstChildOfClass("Humanoid")}
end
local function espScanNpc(inst)
    if inst:IsA("Humanoid") then
        local m = inst.Parent
        if m and m:IsA("Model") and m ~= LP.Character and not Players:GetPlayerFromCharacter(m) then
            ENpcs[m] = true
        end
    end
end
local function espRefresh()
    for model, o in pairs(EObjs) do
        if not model.Parent or not o.hl.Parent or (o.isNpc and not ESP.NPC) or (not o.isNpc and not ESP.Players) then
            espClear(model)
        end
    end
    if ESP.Players then
        for _, plr in ipairs(Players:GetPlayers()) do
            local ch = plr.Character
            if plr ~= LP and ch and not EObjs[ch] then espBuild(ch, plr.DisplayName, false) end
        end
    end
    if ESP.NPC then
        for m in pairs(ENpcs) do
            if not m.Parent then
                ENpcs[m] = nil
            elseif not EObjs[m] then
                espBuild(m, m.Name, true)
            end
        end
    end
end
local function espLabels()
    local root = rootP()
    for _, o in pairs(EObjs) do
        local d = (root and o.head and o.head.Parent) and (root.Position - o.head.Position).Magnitude or 0
        local inRange = d <= ESP.Range
        o.hl.Enabled, o.bb.Enabled = inRange, inRange
        o.hl.FillColor = o.isNpc and NPC_COLOR or ESP_COLORS[ESP.ColorIdx][2]
        if inRange then
            local parts = {}
            if ESP.Name then parts[#parts + 1] = o.name end
            if ESP.HP and o.hum then parts[#parts + 1] = "HP " .. math.floor(o.hum.Health) end
            if ESP.Dist then parts[#parts + 1] = math.floor(d) .. "m" end
            o.lbl.Text = table.concat(parts, "  |  ")
            o.lbl.Visible = #parts > 0
        end
    end
end
local espT1, espT2 = 1, 1
local function espSync()
    if ESP.Players or ESP.NPC then
        espT1, espT2 = 1, 1
        kbind("ESP_LOOP", RunService.Heartbeat, function(dt)
            espT1 = espT1 + dt
            espT2 = espT2 + dt
            if espT1 >= 1 then espT1 = 0 espRefresh() end
            if espT2 >= 0.2 then espT2 = 0 espLabels() end
        end)
    else
        kunbind("ESP_LOOP")
        for m in pairs(EObjs) do espClear(m) end
    end
end

local FlyO = {}
local function clearFly()
    if FlyO.BV then FlyO.BV:Destroy() FlyO.BV = nil end
    if FlyO.BG then FlyO.BG:Destroy() FlyO.BG = nil end
end
local function mvFly(on)
    if on then
        kbind("MV_FLY", RunService.Heartbeat, function()
            local h, root, cam = hum1(), rootP(), Workspace.CurrentCamera
            if not h or not root or not cam then return end
            if not FlyO.BV or FlyO.BV.Parent ~= root then
                clearFly()
                FlyO.BV = new("BodyVelocity", {MaxForce = Vector3.new(1e9, 1e9, 1e9), Velocity = Vector3.zero, Parent = root})
                FlyO.BG = new("BodyGyro", {MaxTorque = Vector3.new(1e9, 1e9, 1e9), P = 9e4, CFrame = root.CFrame, Parent = root})
            end
            h.PlatformStand = true
            local vel, move = Vector3.zero, h.MoveDirection
            if move.Magnitude > 0 then
                local cf = cam.CFrame
                local rel = cf:VectorToObjectSpace(move)
                local dir = cf.RightVector * rel.X - cf.LookVector * rel.Z
                if dir.Magnitude > 0 then vel = dir.Unit * MV.fly end
            end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then vel = vel + Vector3.new(0, MV.fly, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then vel = vel - Vector3.new(0, MV.fly, 0) end
            FlyO.BV.Velocity = vel
            local look = cam.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z)
            if flat.Magnitude > 0.01 then
                FlyO.BG.CFrame = CFrame.lookAt(root.Position, root.Position + flat)
            end
        end)
    else
        kunbind("MV_FLY")
        clearFly()
        local h = hum1()
        if h then h.PlatformStand = false end
    end
end

        kbind("FM_AVOID", RunService.Heartbeat, function(dt)

            for m in pairs(Bots) do

                    Bots[m] = nil
                else
                    local part = m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart
                    if part then
                        local off = root.Position - part.Position
                        if off.Magnitude < FM.avoidD then
                            local flat = Vector3.new(off.X, 0, off.Z)

local Items, Visited = {}, {}

    name = name:lower()
    for _, p in ipairs(COLLECT_PATTERNS) do
        if name:find(p, 1, true) then return true end
    end
    return false
end
local function scanItem(inst)
    if inst:IsA("BasePart") and nameMatches(inst.Name) then
        local ch = LP.Character
        if ch and inst:IsDescendantOf(ch) then return end
        local m = inst:FindFirstAncestorOfClass("Model")
        if not m or not m:FindFirstChildOfClass("Humanoid") then Items[inst] = true end
    end
end
local collectT = 0
local function fmCollect(on)
    if on then
        table.clear(Visited)
        for _, d in ipairs(Workspace:GetDescendants()) do scanItem(d) end
        kbind("FM_ITEM_ADD", Workspace.DescendantAdded, scanItem)
        collectT = 0
        kbind("FM_COLLECT", RunService.Heartbeat, function(dt)
            collectT = collectT + dt
            if collectT < FM.collectV then return end
            collectT = 0
            local root = rootP()
            if not root then return end
            local best, bestD
            local now = os.clock()
            for part in pairs(Items) do
                if not part.Parent then
                    Items[part] = nil
                elseif not Visited[part] or now - Visited[part] > 5 then
                    local d = (part.Position - root.Position).Magnitude
                    if d <= FM.collectR and (not bestD or d < bestD) then best, bestD = part, d end
                end
            end
            if best then
                Visited[best] = now
                root.CFrame = best.CFrame + Vector3.new(0, 2, 0)
                pcall(function()
                    if firetouchinterest then
                        firetouchinterest(root, best, 0)
                        firetouchinterest(root, best, 1)
                    end
                end)
                pcall(function()
                    local prompt = best:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt and fireproximityprompt then fireproximityprompt(prompt) end
                end)
            end
        end)
    else
        kunbind("FM_COLLECT")
        kunbind("FM_ITEM_ADD")
        table.clear(Items)
    end
end

local function isLockName(n)
    if n:lower():find("unlock", 1, true) then return false end
    return (n:find("Lock", 1, true) or n:find("^lock") or n:find("[^%a]lock")) ~= nil
end
local function remember(inst, key, value)
    local o = SHOP.orig[inst]
    if not o then
        o = {}
        SHOP.orig[inst] = o
    end
    if o[key] == nil then
        o[key] = value
        SHOP.count = SHOP.count + 1
    end
end
local LOCK_FALSE = {"Locked", "IsLocked"}
local LOCK_TRUE = {"Unlocked", "IsUnlocked", "Owned", "IsOwned"}

local function unlockInst(inst, seen)
    if seen[inst] then return end
    seen[inst] = true
    if inst:IsA("GuiObject") then
        if inst.Visible and isLockName(inst.Name) then
            remember(inst, "Visible", true)
            inst.Visible = false
        end
        if inst:IsA("GuiButton") then
            if not inst.Active then
                remember(inst, "Active", false)
                inst.Active = true
            end
            pcall(function()
                if inst.Interactable == false then
                    remember(inst, "Interactable", false)
                    inst.Interactable = true
                end
            end)
        end
    end
    for _, a in ipairs(LOCK_FALSE) do
        if inst:GetAttribute(a) == true then
            remember(inst, "attr:" .. a, true)
            inst:SetAttribute(a, false)
        end
    end
    for _, a in ipairs(LOCK_TRUE) do
        if inst:GetAttribute(a) == false then
            remember(inst, "attr:" .. a, false)
            inst:SetAttribute(a, true)
        end
    end
end
local function shopPass()
    local pg = LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local seen = {}
    for _, d in ipairs(pg:GetDescendants()) do
        if (d:IsA("GuiObject") or d:IsA("LayerCollector")) and isShopName(d.Name) and not d:IsDescendantOf(gui) then
            unlockInst(d, seen)
            for _, x in ipairs(d:GetDescendants()) do unlockInst(x, seen) end
        end
    end
end
local function shopRestore()
    for inst, props in pairs(SHOP.orig) do
        if inst.Parent then
            for k, v in pairs(props) do
                pcall(function()
                    if k:sub(1, 5) == "attr:" then inst:SetAttribute(k:sub(6), v) else inst[k] = v end
                end)
            end
        end
    end
    SHOP.orig = {}
    SHOP.count = 0
end
local function shopToggle(on)
    if on then
        pcall(shopPass)
        if SHOP.count > 0 then
            toast(string.format(T("s_done"), SHOP.count), C.RED)
        else
            toast(T("s_none"), C.BLUE)
        end
        if shopVal then shopVal.Text = tostring(SHOP.count) end
        shopT = 0
        kbind("SHOP_LOOP", RunService.Heartbeat, function(dt)
            shopT = shopT + dt
            if shopT < 1.5 then return end
            shopT = 0
            pcall(shopPass)
            if shopVal then shopVal.Text = tostring(SHOP.count) end
        end)
    else
        kunbind("SHOP_LOOP")
        shopRestore()
        if shopVal then shopVal.Text = "0" end
    end
end

------------------------------------------------------------ live loop (FPS / ping / health / position)
local frames, acc = 0, 0
bind(RunService.Heartbeat, function(dt)
    frames = frames + 1
    acc = acc + dt
end)
local function getPing()
    local p
    pcall(function() p = game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() end)
    if not p then pcall(function() p = LP:GetNetworkPing() * 1000 end) end
    return p or 0
end
local startT = os.clock()
local fpsNow = 0
task.spawn(function()
    local n = 0
    while gui.Parent do
        task.wait(0.25)
        n = n + 1
        local inst = (acc > 0) and (frames / acc) or 0
        frames, acc = 0, 0
        fpsNow = (fpsNow == 0) and inst or (fpsNow * 0.4 + inst * 0.6)
        local fps = math.floor(fpsNow + 0.5)
        local ping = math.floor(getPing() + 0.5)
        local hum, hrp = getHum()
        local hp = hum and hum.Health or 0
        local mhp = hum and hum.MaxHealth or 100
        if mhp <= 0 then mhp = 100 end
        local pct = math.clamp(hp / mhp, 0, 1)
        ringFill = math.floor(pct * RING_N + 0.5)

        if n % 4 == 0 then -- push one sample / second into the chart
            table.remove(hist, 1)
            table.insert(hist, fps)
            if holder.Visible and home.Visible then drawChart() end
        end

        if holder.Visible then
            local sess = fmtTime(os.clock() - startT)
            heroClock.Text = nowStr("%H:%M:%S")
            heroSub.Text = nowStr("%d/%m/%Y") .. "  ·  " .. sess
            if home.Visible then
                stPlayers.Text = tostring(#Players:GetPlayers())
                stFps.Text = tostring(fps)
                stPing.Text = ping .. " ms"
                fpsTop.Text = tostring(fps)
                ringPct.Text = math.floor(pct * 100 + 0.5) .. "%"
                legendVals[1].Text = math.floor(hp) .. "/" .. math.floor(mhp)
                legendVals[2].Text = tostring(fps)
                legendVals[3].Text = ping .. "ms"
            end
            if pages.Me.Visible then
                hpText.Text = math.floor(hp) .. " / " .. math.floor(mhp)
                tw(hpFill, 0.25, {Size = UDim2.fromScale(pct, 1)})
                tFps.Text = tostring(fps)
                tPing.Text = ping .. " ms"
                tSpeed.Text = hum and tostring(math.floor(hum.WalkSpeed)) or "-"
                tSess.Text = sess
                if hrp then
                    local p = hrp.Position
                    infoVals.position.Text = math.floor(p.X) .. ", " .. math.floor(p.Y) .. ", " .. math.floor(p.Z)
                else
                    infoVals.position.Text = "-"
                end
                infoVals.in_server.Text = #Players:GetPlayers() .. " / " .. Players.MaxPlayers
                infoVals.acc_age.Text = LP.AccountAge .. " " .. T("days")
                infoVals.server.Text = (game.JobId ~= "" and string.sub(game.JobId, 1, 8)) or "Studio"
            end
        end
    end
end)
