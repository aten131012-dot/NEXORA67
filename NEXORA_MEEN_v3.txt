local function new(cls, props)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    return o
end
local function tw(o, t, p, style, dir)
    local tn = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), p)
    tn:Play()
    return tn
end
local function corner(p, r)
    return new("UICorner", {CornerRadius = UDim.new(0, r or 8), Parent = p})
end
local function stroke(p, col, th, tr)
    return new("UIStroke", {Color = col or C.LINE, Thickness = th or 1, Transparency = tr or 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = p})
end
local function txt(parent, props)
    local d = {BackgroundTransparency = 1, Font = Enum.Font.Gotham, TextSize = 13, TextColor3 = C.TEXT,
        TextXAlignment = Enum.TextXAlignment.Left, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    return new("TextLabel", d)
end
local function btn(parent, props)
    local d = {AutoButtonColor = false, Text = "", BorderSizePixel = 0, Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = C.TEXT, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    return new("TextButton", d)
end
local function copyText(s)
    local fn = (typeof(setclipboard) == "function" and setclipboard) or (typeof(toclipboard) == "function" and toclipboard)
    if not fn then return false end
    return (pcall(fn, s))
end
local function fmtTime(s)
    s = math.max(0, math.floor(s))
    return string.format("%02d:%02d:%02d", math.floor(s / 3600), math.floor(s / 60) % 60, s % 60)
end

end

local function makeDraggable(handle, target)
    local dragging, start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, start, startPos = true, i.Position, target.Position
            target:SetAttribute("moved", false)
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - start
            if d.Magnitude > 5 then target:SetAttribute("moved", true) end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function tabIcon(parent, kind)
    local box = new("Frame", {Position = UDim2.fromOffset(14, 15), Size = UDim2.fromOffset(16, 16), BackgroundTransparency = 1, ZIndex = 3, Parent = parent})
    local parts, rings = {}, {}
    local function sq(x, y, w, h, r)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h), BackgroundColor3 = C.MUTED, BorderSizePixel = 0, ZIndex = 3, Parent = box})
        corner(f, r or 2)
        table.insert(parts, f)
        return f
    end
    local function ringF(x, y, d, th)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(d, d), BackgroundTransparency = 1, ZIndex = 3, Parent = box})
        new("UICorner", {CornerRadius = UDim.new(0.5, 0), Parent = f})
        local st = new("UIStroke", {Color = C.MUTED, Thickness = th, Parent = f})
        table.insert(rings, st)
    end
    if kind == "Home" then
        sq(2, 7, 12, 9, 2)
        local rf = sq(3, 1, 10, 10, 1) rf.Rotation = 45
    elseif kind == "Farm" then
        local h = sq(7, 1, 2, 15, 1) h.Rotation = 35
        local hd = sq(0, 1, 15, 3, 1) hd.Rotation = -35
    elseif kind == "Shop" then
        sq(1, 6, 14, 10, 2) ringF(4, 0, 8, 2)
    elseif kind == "Players" then
        ringF(1, 0, 6, 2) ringF(9, 0, 6, 2) sq(0, 9, 7, 7, 3) sq(9, 9, 7, 7, 3)
    elseif kind == "Egg" then
        sq(3, 0, 10, 16, 5)
    else -- Settings (gear)
        ringF(3, 3, 10, 2)
        sq(7, 0, 2, 3, 1) sq(7, 13, 2, 3, 1) sq(0, 7, 3, 2, 1) sq(13, 7, 3, 2, 1)
    end
    return function(col)
        for _, p in ipairs(parts) do p.BackgroundColor3 = col end
        for _, r in ipairs(rings) do r.Color = col end
    end
end

local function setPage(n)
    currentPage = n
    for k, p in pairs(pages) do p.Visible = (k == n) end
    for k, b in pairs(tabBtns) do
        local on = (k == n)
        tw(b, 0.2, {BackgroundColor3 = on and C.BLUE or C.TAB})
        local lbl = b:FindFirstChild("L")
        if lbl then lbl.TextColor3 = on and WHITE or C.MUTED end
        tabIcons[k](on and WHITE or C.MUTED)
    end
end

local function addTab(key, title, order)
    local b = btn(tabList, {Size = UDim2.new(1, 0, 0, 46), BackgroundColor3 = C.TAB, LayoutOrder = order})
    corner(b, 10)
    txt(b, {Name = "L", Position = UDim2.fromOffset(44, 0), Size = UDim2.new(1, -50, 1, 0), Text = title, Font = Enum.Font.GothamMedium, TextSize = 14, TextColor3 = C.MUTED, ZIndex = 3})
    tabIcons[key] = tabIcon(b, key)
    tabBtns[key] = b
    b.MouseButton1Click:Connect(function() setPage(key) end)
    local p = new("ScrollingFrame", {Name = key, Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3,
        ScrollBarImageColor3 = C.BLUE, CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, Visible = false, Parent = content})
    new("UIListLayout", {Padding = UDim.new(0, 10), SortOrder = Enum.SortOrder.LayoutOrder, Parent = p})
    new("UIPadding", {PaddingRight = UDim.new(0, 6), PaddingBottom = UDim.new(0, 8), Parent = p})
    pages[key] = p
    return p
end

    if type(v) ~= "string" then return 0 end

    local mult = ({ k = 1e3, m = 1e6, b = 1e9, t = 1e12, qa = 1e15, qi = 1e18 })[(suf or ""):lower()] or 1
    return num * mult

local function walkTo(sys, id, getPos, stopDist, speed)
    while alive(sys, id) do
        if not gateWait(sys, id) then return false end
        local root, target = eRoot(), getPos()
        if not root or not target then return false end
        local p = root.Position
        local flat = Vector3.new(target.X - p.X, 0, target.Z - p.Z)
        local dist = flat.Magnitude
        local look = Vector3.new(target.X, CFG.GroundY, target.Z)
        if dist <= stopDist or dist < 0.01 then
            local here = Vector3.new(p.X, CFG.GroundY, p.Z)
            if (look - here).Magnitude > 0.01 then root.CFrame = CFrame.lookAt(here, look) end
            root.AssemblyLinearVelocity = Vector3.zero
            return true
        end
        local dt = math.min(RunService.Heartbeat:Wait(), CFG.MaxDt)
        local step = math.min(speed * dt, dist - stopDist)
        local np = Vector3.new(p.X, CFG.GroundY, p.Z) + flat.Unit * step
        root.CFrame = CFrame.lookAt(np, look)
        root.AssemblyLinearVelocity = Vector3.zero
    end
    return false
end

local function signature(btn)
    local parts = {}
    local function add(k, v) parts[#parts + 1] = k .. "=" .. tostring(v) end
    add("vis", btn.Visible); add("bg", tostring(btn.BackgroundColor3) .. "/" .. btn.BackgroundTransparency)
    if btn:IsA("TextButton") then add("txt", btn.Text); add("tc", btn.TextColor3) end
    for _, d in ipairs(btn:GetDescendants()) do
        if d:IsA("TextLabel") then add("t:" .. d.Name, d.Text .. "|" .. tostring(d.TextColor3) .. "|" .. d.TextTransparency)
        elseif d:IsA("ImageLabel") then add("i:" .. d.Name, d.Image .. "|" .. tostring(d.ImageColor3) .. "|" .. d.ImageTransparency)
        elseif d:IsA("UIStroke") then add("s:" .. d.Name, tostring(d.Color))
        elseif d:IsA("UIGradient") then local kp = d.Color.Keypoints; add("g:" .. d.Name, tostring(kp[1].Value) .. tostring(kp[#kp].Value))
        elseif d:IsA("BoolValue") then add("b:" .. d.Name, d.Value)
        elseif d:IsA("NumberValue") then add("n:" .. d.Name, d.Value) end
    end
    return table.concat(parts, ";")
end
local function swingState(btn) -- attributes -> BoolValue -> ON/OFF text -> learned look
    for _, a in ipairs(STATE_ATTRS) do local v = btn:GetAttribute(a); if type(v) == "boolean" then return v end end
    for _, c in ipairs(btn:GetChildren()) do if c:IsA("BoolValue") then return c.Value end end
    local texts = {}
    if btn:IsA("TextButton") then texts[#texts + 1] = btn.Text end
    for _, d in ipairs(btn:GetDescendants()) do if d:IsA("TextLabel") then texts[#texts + 1] = d.Text end end
    for _, t in ipairs(texts) do
        local u = t:upper()
        if u:find("%f[%a]OFF%f[%A]") then return false end
        if u:find("%f[%a]ON%f[%A]") then return true end
    end
    if Swing.learned then
        local s = signature(btn)
        if s == Swing.learned.on then return true end
        if s == Swing.learned.off then return false end
    end
    return nil
end
local function pressSwing(btn)
    local before, known = signature(btn), swingState(btn)
    pcall(function() if firesignal then firesignal(btn.MouseButton1Click) end end)
    pcall(function() if firesignal then firesignal(btn.Activated) end end)
    task.wait(0.35)
    if not Swing.learned and known == nil and not Swing.pressed then
        local after = signature(btn)
        if after ~= before then Swing.learned = { off = before, on = after } end -- first press: OFF -> ON
    end
    Swing.pressed = true
end
local function ensurePickaxe()
    local char, bp = LP.Character, LP:FindFirstChildOfClass("Backpack")
    for _, holder in ipairs({ char, bp }) do
        if holder then
            for _, t in ipairs(holder:GetChildren()) do
                if t:IsA("Tool") and t.Name:lower():find(CFG.PickaxePattern) then
                    if holder ~= char then local h = eHum(); if h then pcall(function() h:EquipTool(t) end) end end
                    return t
                end
            end
        end
    end
end
local function swingMonitor(id)
    local warned = false
    while Swing.refs > 0 and Swing.monitorJob == id do
        local ok, err = pcall(function()
            local tool, btn = ensurePickaxe(), swingButton()
            if btn then
                warned = false
                local st = swingState(btn)
                if st == false or (st == nil and not Swing.pressed) then pressSwing(btn) end
            else
                if not warned then toast(T("swing_nobtn"), C.RED); warned = true end
                if tool then pcall(function() tool:Activate() end) end -- fallback: the game's swing
            end
        end)
        if not ok then warn("[SwingMonitor] " .. tostring(err)) end
        task.wait(0.4)
    end
end
function Swing.acquire()
    Swing.refs = Swing.refs + 1
    if Swing.refs == 1 then Swing.monitorJob = Swing.monitorJob + 1; task.spawn(swingMonitor, Swing.monitorJob) end
end
function Swing.release()
    Swing.refs = math.max(0, Swing.refs - 1)
    if Swing.refs == 0 then Swing.monitorJob = Swing.monitorJob + 1 end
end

local function pickBreakTarget()
    local root = eRoot(); if not root then return end
    local best
    for _, e in ipairs(scanEggs()) do
        if e.rarity and Settings.SW.Rarity[e.rarity] then
            e.dist = (e.part.Position - root.Position).Magnitude
            e.rank = RANK[e.rarity] or 0
            if not best or e.size > best.size
               or (e.size == best.size and (e.rank > best.rank or (e.rank == best.rank and e.dist < best.dist))) then best = e end
        end
    end
    return best
end

local function runBreak(id)
    Swing.acquire()
    local held = true
    local ok, err = pcall(function()
        while alive(SW, id) do
            if not gateWait(SW, id) then break end
            local egg = pickBreakTarget()
            if not egg then
                say("SW", "b_none"); task.wait(1.5)
            else
                say("SW", "b_walk", RARITY_TH[egg.rarity] and (LANG == "th" and RARITY_TH[egg.rarity] or egg.rarity) or egg.rarity, egg.size)
                local stop = math.max(CFG.BreakStopDist, egg.half + CFG.BreakGap)
                local arrived = walkTo(SW, id, function() return eggAlive(egg) and egg.part.Position or nil end, stop, Settings.SW.Speed)
                if arrived and alive(SW, id) then
                    say("SW", "b_break")
                    local t0 = os.clock()
                    while alive(SW, id) and eggAlive(egg) and os.clock() - t0 < 60 do
                        if not gateWait(SW, id) then break end
                        local root = eRoot()
                        if root and egg.part.Parent then
                            local p, t = root.Position, egg.part.Position
                            if Vector3.new(t.X - p.X, 0, t.Z - p.Z).Magnitude > stop + 3 then break end -- drifted -> re-walk
                            root.CFrame = CFrame.lookAt(Vector3.new(p.X, CFG.GroundY, p.Z), Vector3.new(t.X, CFG.GroundY, t.Z))
                            root.AssemblyLinearVelocity = Vector3.zero
                        end
                        task.wait(0.1)
                    end
                    if not eggAlive(egg) then say("SW", "b_done") end
                end
                task.wait(0.1)
            end
        end
    end)
    if held then held = false; Swing.release() end
    if not ok then warn("[AutoBreak] " .. tostring(err)) end
    return ok
end

local Busy = { conns = {}, collide = {}, saved = nil, bound = false }
local LOCK_KEYS = { Enum.KeyCode.W, Enum.KeyCode.A, Enum.KeyCode.S, Enum.KeyCode.D, Enum.KeyCode.Space,
    Enum.KeyCode.Up, Enum.KeyCode.Down, Enum.KeyCode.Left, Enum.KeyCode.Right, Enum.KeyCode.Thumbstick1 }

local function setBusy(on)
    if on == Settings.ST.Busy then return end
    Settings.ST.Busy = on
    EventGate.StealBusy = on
    if on then
        pcall(function() -- 1) swallow movement keys at very high priority
            CAS:BindActionAtPriority("ST_Lock", function() return Enum.ContextActionResult.Sink end, false,
                Enum.ContextActionPriority.High.Value + 1000, unpack(LOCK_KEYS))
            Busy.bound = true
        end)
        local hum = eHum() -- 2) freeze the humanoid
        if hum then
            Busy.saved = { ws = hum.WalkSpeed, jp = hum.JumpPower, jh = hum.JumpHeight, ujp = hum.UseJumpPower }
            hum.WalkSpeed = 0; hum.JumpPower = 0; hum.JumpHeight = 0
            if Settings.ST.AntiRagdoll then
                pcall(function()
                    hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                    hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                end)
            end
        end
        Busy.conns[#Busy.conns + 1] = RunService.Stepped:Connect(function() -- 3) noclip + anti-ragdoll
            local char = LP.Character; if not char then return end
            for _, d in ipairs(char:GetDescendants()) do
                if Settings.ST.Noclip and d:IsA("BasePart") then
                    if d.CanCollide then Busy.collide[d] = true; d.CanCollide = false end
                end
                if Settings.ST.AntiRagdoll then
                    if d:IsA("Motor6D") and not d.Enabled then d.Enabled = true
                    elseif d:IsA("BallSocketConstraint") and d.Enabled then d.Enabled = false end
                end
            end
        end)
    else
        for _, c in ipairs(Busy.conns) do c:Disconnect() end
        Busy.conns = {}
        if Busy.bound then pcall(function() CAS:UnbindAction("ST_Lock") end); Busy.bound = false end
        for part in pairs(Busy.collide) do if part and part.Parent then part.CanCollide = true end end
        Busy.collide = {}
        local hum = eHum()
        if hum and Busy.saved then
            hum.WalkSpeed = Busy.saved.ws; hum.UseJumpPower = Busy.saved.ujp
            hum.JumpPower = Busy.saved.jp; hum.JumpHeight = Busy.saved.jh
            pcall(function()
                hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
                hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
            end)
        end
        Busy.saved = nil
    end
end

local function pickupsFolder() return Workspace:FindFirstChild("AnimalPickups") end
local function animalIncome(inst)
    local c = inst:GetAttribute("CashPerSecond")
    if type(c) == "number" then return c end
    local ov = inst:FindFirstChild("AnimalOverhead", true)
    local lab = ov and ov:FindFirstChild("CashPerSecond") and ov.CashPerSecond:FindFirstChild("CashPerSecond")
    if lab and lab:IsA("TextLabel") then
        local n = parseNumber((lab.Text:gsub("[^%d%.%a]", ""):gsub("[sS]$", "")))
        if n > 0 then return n end
    end
    for k, v in pairs(inst:GetAttributes()) do
        local lk = k:lower()
        if lk:find("income") or lk:find("persecond") or lk:find("cashper") then return parseNumber(v) end
    end
    return 0
end
local function animalPrompt(inst)
    local p = inst:FindFirstChildWhichIsA("ProximityPrompt", true)
    if p then return p end
    local pos, best, bd = posOf(inst), nil, 15 -- prompts may be created at runtime on a nearby anchor
    if not pos then return end
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Parent then
            local holder = d.Parent:IsA("Attachment") and d.Parent.Parent or d.Parent
            local pp = holder and posOf(holder)
            if pp and (pp - pos).Magnitude < bd then best, bd = d, (pp - pos).Magnitude end
        end
    end
    return best
end
local function listAnimals()
    local out, folder = {}, pickupsFolder()
    if not folder then return out end
    for _, a in ipairs(folder:GetChildren()) do
        local pos = posOf(a)
        if pos then
            local mut = a:GetAttribute("Mutation")
            if mut == nil then
                local ml = a:FindFirstChild("Mutation", true)
                if ml and ml:IsA("TextLabel") and ml.Visible and ml.Text ~= "" then mut = ml.Text end
            end
            out[#out + 1] = { inst = a, pos = pos, rarity = readRarity(a), income = animalIncome(a),
                mutation = (type(mut) == "string" and mut ~= "") and mut or "None" }
        end
    end
    return out
end
local function anySet(t) return next(t) ~= nil end
local function passesFilter(a)
    local s = Settings.ST
    if anySet(s.Rarity) and not (a.rarity and s.Rarity[a.rarity]) then return false end
    if anySet(s.Mutation) and not s.Mutation[a.mutation:lower()] then return false end
    return a.income >= s.MinIncome
end
local function pickStealTarget(skip)
    local root = eRoot(); if not root then return end
    local best
    for _, a in ipairs(listAnimals()) do
        if passesFilter(a) and not (skip and skip[a.inst]) then
            a.dist = (a.pos - root.Position).Magnitude
            a.rank = RANK[a.rarity or ""] or 0
            if not best or a.income > best.income or (a.income == best.income and (a.rank > best.rank
               or (a.rank == best.rank and a.dist < best.dist))) then best = a end
        end
    end
    return best
end
local function isCarrying()
    local char = LP.Character; if not char then return false end
    if char:GetAttribute("Carrying") or LP:GetAttribute("Carrying") then return true end
    local carried = Workspace:FindFirstChild("CarriedAnimals")
    if carried then
        for _, c in ipairs(carried:GetChildren()) do
            if c:GetAttribute("OwnerUserId") == LP.UserId or c:GetAttribute("CarrierUserId") == LP.UserId
               or c.Name:find(LP.Name, 1, true) then return true end
        end
    end
    for _, t in ipairs(char:GetChildren()) do
        if t:IsA("Tool") and not t.Name:lower():find(CFG.PickaxePattern) then return true end
    end
    return false
end

local function goSafe(id, speed)
    local p = safeZonePos()
    if not p then say("ST", "s_nosafe"); return false end
    return walkTo(ST, id, function() return p end, 3, speed)

-- smallest egg in the current zone (same size -> nearer); optional egg-rarity filter
local function smallestEggInZone()
    local root = eRoot(); if not root then return end
    local eggs = scanEggs()
    local nearest, nd = nil, math.huge
    for _, e in ipairs(eggs) do
        local d = (e.part.Position - root.Position).Magnitude
        if d < nd then nearest, nd = e, d end
    end
    if not nearest then return end
    local best
    for _, e in ipairs(eggs) do
        if e.zone == nearest.zone and (not anySet(Settings.ST.EggRarity) or (e.rarity and Settings.ST.EggRarity[e.rarity])) then
            e.dist = (e.part.Position - root.Position).Magnitude
            if not best or e.size < best.size or (e.size == best.size and e.dist < best.dist) then best = e end
        end
    end
    return best
end

-- STEP 1 of "dig then steal": walk to the egg and break it. returns true when broken
local function digEgg(id, egg)
    local s = Settings.ST
    local stop = math.max(CFG.BreakStopDist, egg.half + CFG.BreakGap)
    say("ST", s.TpMode and "tp_walk" or "s_dig_walk")
    if not walkTo(ST, id, function() return eggAlive(egg) and egg.part.Position or nil end, stop, s.Speed) then return false end
    say("ST", s.TpMode and "tp_break" or "s_dig_break")
    Swing.acquire()
    local ok, err = pcall(function()
        local t0 = os.clock()
        while alive(ST, id) and eggAlive(egg) and os.clock() - t0 < 60 do
            if not gateWait(ST, id) then break end
            local root = eRoot()
            if root and egg.part.Parent then
                local p, t = root.Position, egg.part.Position
                root.CFrame = CFrame.lookAt(Vector3.new(p.X, CFG.GroundY, p.Z), Vector3.new(t.X, CFG.GroundY, t.Z))
                root.AssemblyLinearVelocity = Vector3.zero
            end
            task.wait(0.15)
        end
    end)
    Swing.release()
    if not ok then error(err, 0) end
    return not eggAlive(egg)
end

-- STEP 2: wait for the animal that dropped at the broken egg (closest one that passes the filters)
local function waitDrop(id, pos)
    say("ST", "s_wait_drop")
    for _ = 1, 20 do
        if not alive(ST, id) then return end
        local best, bd = nil, 30
        for _, a in ipairs(listAnimals()) do
            local d = (a.pos - pos).Magnitude
            if d < bd and passesFilter(a) then best, bd = a, d end
        end
        if best then return best end
        task.wait(0.3)
    end
end

-- STEP 3: grab it
local function grab(id, a)
    local s = Settings.ST
    say("ST", "s_walk")
    if not walkTo(ST, id, function() return a.inst.Parent and posOf(a.inst) or nil end, 4, s.Speed) then return "lost" end
    local prompt = animalPrompt(a.inst)
    if not prompt then say("ST", "s_noprompt"); return "lost" end
    say("ST", "s_steal")
    for _ = 1, 5 do
        if not alive(ST, id) then return "lost" end
        firePrompt(prompt); task.wait(0.4)
        if isCarrying() or not a.inst.Parent then s.Collected = s.Collected + 1; return "ok" end
    end
    return "lost"
end

-- STEP 4: take the guard hit, then carry on
local function handleGuard(id)
    local g = guardInst()
    if not g then return end
    say("ST", "s_guard")
    walkTo(ST, id, function() return g.Parent and posOf(g:FindFirstChild("BodyHitbox") or g) or nil end, 5, Settings.ST.Speed)
    task.wait(0.8)
end

-- STEP 5: deliver at START (TP mode: teleport with retries, fallback to walking); re-pick if dropped
local function deliver(id, a)
    local s = Settings.ST
    if s.TpMode then
        local sp = startPos() or safeZonePos()
        local okTp = false
        if sp then
            say("ST", "tp_go")
            for _ = 1, 3 do
                teleportTo(sp); task.wait(0.4)
                local r = eRoot()
                if r and (Vector3.new(r.Position.X, 0, r.Position.Z) - Vector3.new(sp.X, 0, sp.Z)).Magnitude < 12 then okTp = true; break end
            end
        end
        if okTp then task.wait(0.5); return true end
        say("ST", "tp_fail")
    end
    for _ = 1, 3 do
        if not alive(ST, id) then return false end
        local sp = startPos()
        if not sp then say("ST", "s_nostart"); return false end
        say("ST", "s_deliver")
        local arrived = walkTo(ST, id, function() if not isCarrying() then return nil end return sp end, 3, s.Speed)
        if arrived then task.wait(0.5); return true end
        if not isCarrying() then
            if a.inst.Parent and animalPrompt(a.inst) then
                say("ST", "s_dropped")
                if grab(id, a) ~= "ok" then return false end
                s.Collected = s.Collected - 1 -- do not double count a re-pickup
            else
                say("ST", "s_lost"); return false
            end
        end
    end
    return false
end

local function runSteal(id)
    local s = Settings.ST
    local ok, err = pcall(function()
        local origin = eRoot() and eRoot().Position
        say("ST", "s_wait")
        while alive(ST, id) and not pickupsFolder() do task.wait(0.5) end
        local skip = {}
        while alive(ST, id) do
            if not gateWait(ST, id) then break end
            if s.SafeWait then say("ST", "s_safewait"); goSafe(id, s.Speed); task.wait(1) end
            if not alive(ST, id) then break end
            setBusy(true)
            local target
            if not s.DigFirst then target = pickStealTarget(skip) end
            if not target then
                -- DIG THEN STEAL: break the smallest egg of this zone, then grab what drops
                local egg = smallestEggInZone()
                if egg then
                    local broke = digEgg(id, egg)
                    if broke and alive(ST, id) then
                        local at = egg.part.Parent and egg.part.Position or egg.model:GetPivot().Position
                        target = waitDrop(id, at)
                    end
                end
                if not target then target = pickStealTarget(skip) end
            end
            local result = "lost"
            if target then
                result = grab(id, target)
                if result == "ok" then
                    handleGuard(id)
                    if not deliver(id, target) then result = "lost" end
                end
            else
                say("ST", "s_safe")
            end
            setBusy(false)
            if target and result ~= "ok" then skip[target.inst] = true end
            if alive(ST, id) then
                if target then
                    say("ST", "s_return")
                    if origin then walkTo(ST, id, function() return origin end, 3, s.Speed) else goSafe(id, s.Speed) end
                else
                    goSafe(id, s.Speed); task.wait(2)
                end
            end
            task.wait(0.2)
        end
    end)
    setBusy(false) -- never leave controls locked, even after an error
    if not ok then warn("[AutoSteal] " .. tostring(err)) end
    return ok
end

------------------------------------------------ start / stop
local function stopSystem(sys)
    sys.Running = false
    sys.Job = sys.Job + 1
    if sys == ST then setBusy(false) end
    local nm = (sys == SW) and "SW" or "ST"
    if Eng.resetToggle[nm] then Eng.resetToggle[nm](false) end
end
local function startSystem(sys)
    local nm = (sys == SW) and "SW" or "ST"
    if sys == SW and not anySet(Settings.SW.Rarity) then
        toast(T("b_norar"), C.RED); say("SW", "b_norar")
        if Eng.resetToggle.SW then Eng.resetToggle.SW(false) end
        return
    end
    sys.Job = sys.Job + 1
    sys.Running = true
    local id = sys.Job
    task.spawn(function()
        local ok
        if sys == SW then ok = runBreak(id) else ok = runSteal(id) end
        if sys.Job == id and sys.Running then stopSystem(sys) end -- error / natural exit -> auto-stop + reset switch
        say(nm, ok and "stopped" or "err_stop")
    end)
end

end)

return Eng

build("ผู้เล่น", function()
    W.section(pPlayers, "เครื่องมือผู้เล่น", 1)
    local plrEsp, plrHl = false, {}
    local function clearPlrEsp() for _, h in pairs(plrHl) do h:Destroy() end plrHl = {} end
    local function updPlrEsp()
        if not plrEsp then clearPlrEsp() return end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local h = plrHl[p]
                if not h or not h.Parent then
                    h = new("Highlight", {FillColor = C.RED, OutlineColor = WHITE, FillTransparency = 0.7, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop, Parent = gui})
                    plrHl[p] = h
                end
                h.Adornee = p.Character
            end
        end
        for p, h in pairs(plrHl) do if not p.Parent then h:Destroy() plrHl[p] = nil end end
    end
    W.toggle(pPlayers, "ไฮไลต์ผู้เล่น (ESP)", "มองเห็นผู้เล่นทะลุกำแพง", false, 2, function(v) plrEsp = v updPlrEsp() end)
    local specTarget
    local function stopSpec()
        specTarget = nil
        local h = getHum()
        local cam = workspace.CurrentCamera
        if cam and h then cam.CameraSubject = h end
    end
    W.button(pPlayers, "หยุดดูผู้เล่น", "กลับมามุมมองตัวเอง", "กลับ", 3, stopSpec)
    W.section(pPlayers, "รายชื่อผู้เล่นในเซิร์ฟ", 4)

    local rows = {}
    local function rebuild()
        for _, r in ipairs(rows) do r:Destroy() end
        rows = {}
        local i = 0
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP then
                i = i + 1
                local f = new("Frame", {Size = UDim2.new(1, 0, 0, 52), BackgroundColor3 = C.ROW, BorderSizePixel = 0, LayoutOrder = 100 + i, Parent = pPlayers})
                corner(f, 12)
                stroke(f, C.LINE, 1, 0.35)
                rows[#rows + 1] = f
                txt(f, {Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -170, 1, 0), Text = p.DisplayName .. "  (@" .. p.Name .. ")", TextSize = 12,
                    Font = Enum.Font.GothamMedium, TextTruncate = Enum.TextTruncate.AtEnd})
                local bt = btn(f, {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -78, 0.5, 0), Size = UDim2.fromOffset(60, 30), BackgroundColor3 = C.BLUE, Text = "วาร์ป", TextSize = 12})
                corner(bt, 9)
                bt.MouseButton1Click:Connect(function()
                    local r = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                    if r then tpNear(r.Position) else toast("ผู้เล่นคนนี้ไม่อยู่ในเกม", C.RED) end
                end)
                local bv = btn(f, {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.fromOffset(60, 30), BackgroundColor3 = C.OFF, Text = "ดู", TextSize = 12})
                corner(bv, 9)
                bv.MouseButton1Click:Connect(function()
                    local h = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                    local cam = workspace.CurrentCamera
                    if h and cam then specTarget = p cam.CameraSubject = h else toast("ผู้เล่นคนนี้ไม่อยู่ในเกม", C.RED) end
                end)
            end
        end
        if i == 0 then
            local f = new("Frame", {Size = UDim2.new(1, 0, 0, 44), BackgroundColor3 = C.ROW, BorderSizePixel = 0, LayoutOrder = 101, Parent = pPlayers})
            corner(f, 12)
            rows[1] = f
            txt(f, {Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -28, 1, 0), Text = "ไม่มีผู้เล่นคนอื่นในเซิร์ฟ", TextColor3 = C.MUTED})
        end
        updPlrEsp()
    end
    rebuild()
    bind(Players.PlayerAdded, function() task.defer(rebuild) end)
    bind(Players.PlayerRemoving, function(p) if specTarget == p then stopSpec() end task.defer(rebuild) end)
    task.spawn(function() while gui.Parent do pcall(updPlrEsp) task.wait(1.5) end end)
    gui.Destroying:Connect(function() clearPlrEsp() pcall(stopSpec) end)
end)

build("ไข่", function()
    W.section(pEgg, "มองเห็นไข่ (ESP)", 1)
    local espSet, espOn, espObjs = {}, false, {}
    local function clearEsp() for _, b in pairs(espObjs) do b:Destroy() end espObjs = {} end
    W.toggle(pEgg, "แสดง ESP ไข่", "แสดงระดับและขนาดไข่เหนือไข่แต่ละใบ", false, 2, function(v) espOn = v if not v then clearEsp() end end)
    W.dropdown(pEgg, "ระดับที่แสดง / ใช้วาร์ป (ไม่เลือก = ทุกระดับ)", Eng.RARITIES, espSet, 3, "ทุกระดับ", nil, nil)
    W.section(pEgg, "สถานะไข่ในแมพ", 4)
    local eTotal = W.info(pEgg, "ไข่ทั้งหมดในแมพ", 5)
    local eMatch = W.info(pEgg, "ตรงเงื่อนไข", 6)
    local eList = W.info(pEgg, "แยกตามระดับ", 7)
    W.section(pEgg, "วาร์ปด่วน", 8)
    local function eggTarget(smallest)
        local _, root = getHum()
        local best
        for _, e in ipairs(Eng.scanEggs()) do
            if not next(espSet) or (e.rarity and espSet[e.rarity]) then
                e.rank = Eng.RANK[e.rarity or ""] or 0
                e.dist = root and (e.part.Position - root.Position).Magnitude or 0
                local better
                if not best then better = true
                elseif smallest then better = e.size < best.size or (e.size == best.size and e.dist < best.dist)
                else better = e.size > best.size or (e.size == best.size and (e.rank > best.rank or (e.rank == best.rank and e.dist < best.dist))) end
                if better then best = e end
            end
        end
        return best
    end
    W.button(pEgg, "ไปไข่ที่ดีที่สุด", "ใหญ่ที่สุด / หายากที่สุดในระดับที่เลือก", "วาร์ป", 9, function()
        local e = eggTarget(false)
        if e then tpNear(e.part.Position) else toast(T("e_none"), C.RED) end
    end)
    W.button(pEgg, "ไปไข่ที่เล็กที่สุด", "ไข่ใบเล็กสุด (ทุบง่ายสุด)", "วาร์ป", 10, function()
        local e = eggTarget(true)
        if e then tpNear(e.part.Position) else toast(T("e_none"), C.RED) end
    end)
    W.button(pEgg, "ไปจุดปลอดภัย (Safe Zone)", nil, "วาร์ป", 11, function()
        local p = Eng.safeZonePos()
        if p then tpNear(p) else toast(T("s_nosafe"), C.RED) end
    end)

    task.spawn(function()
        while gui.Parent do
            pcall(function()
                local eggs = Eng.scanEggs()
                if espOn then
                    local seen = {}
                    for _, e in ipairs(eggs) do
                        if not next(espSet) or (e.rarity and espSet[e.rarity]) then
                            seen[e.part] = true
                            local b = espObjs[e.part]
                            if not b then
                                b = new("BillboardGui", {Adornee = e.part, AlwaysOnTop = true, Size = UDim2.fromOffset(140, 30), StudsOffset = Vector3.new(0, e.half + 2, 0), Parent = gui})
                                txt(b, {Name = "L", Size = UDim2.fromScale(1, 1), TextXAlignment = Enum.TextXAlignment.Center, Font = Enum.Font.GothamBold, TextSize = 13, TextStrokeTransparency = 0.3})
                                espObjs[e.part] = b
                            end
                            local l = b:FindFirstChild("L")
                            local rn = e.rarity or "?"
                            l.Text = (Eng.RARITY_TH[rn] or rn) .. "  ·  " .. e.size
                            l.TextColor3 = Eng.RARITY_COL[rn] or WHITE
                        end
                    end
                    for p, b in pairs(espObjs) do if not seen[p] then b:Destroy() espObjs[p] = nil end end
                end
                if pEgg.Visible then
                    local match, byR = 0, {}
                    for _, e in ipairs(eggs) do
                        if not next(espSet) or (e.rarity and espSet[e.rarity]) then match = match + 1 end
                        byR[e.rarity or "?"] = (byR[e.rarity or "?"] or 0) + 1
                    end
                    eTotal.Text = tostring(#eggs)
                    eMatch.Text = tostring(match)
                    local parts = {}
                    for i = #Eng.RARITIES, 1, -1 do
                        local r = Eng.RARITIES[i]
                        if byR[r] then parts[#parts + 1] = (Eng.RARITY_TH[r] or r) .. "×" .. byR[r] end
                    end
                    eList.Text = (#parts > 0) and table.concat(parts, " ") or "-"
                end
            end)
            task.wait(1.2)
        end
    end)
    gui.Destroying:Connect(clearEsp)
end)

build("ตั้งค่า", function()
    local ORIG = {}
    pcall(function()
        ORIG.GlobalShadows = Lighting.GlobalShadows
        ORIG.Brightness = Lighting.Brightness
        ORIG.ClockTime = Lighting.ClockTime
        ORIG.FogEnd = Lighting.FogEnd
        ORIG.Ambient = Lighting.Ambient
        ORIG.OutdoorAmbient = Lighting.OutdoorAmbient
    end)
    local ORIGQ
    pcall(function() ORIGQ = settings().Rendering.QualityLevel end)
    local st = { boost = false, full = false, ws = nil, jp = nil }
    local fullApplied = false
    local function applyLighting()
        pcall(function() Lighting.GlobalShadows = (ORIG.GlobalShadows == true) and not (st.boost or st.full) end)
        pcall(function()
            if st.full then
                fullApplied = true
                Lighting.Brightness = 2
                Lighting.ClockTime = 14
                Lighting.FogEnd = 100000
                Lighting.Ambient = Color3.fromRGB(178, 178, 178)
                Lighting.OutdoorAmbient = Color3.fromRGB(178, 178, 178)
            elseif fullApplied then
                fullApplied = false
                Lighting.Brightness = ORIG.Brightness
                Lighting.ClockTime = ORIG.ClockTime
                Lighting.FogEnd = ORIG.FogEnd
                Lighting.Ambient = ORIG.Ambient
                Lighting.OutdoorAmbient = ORIG.OutdoorAmbient
            end
        end)
        pcall(function()
            if st.boost then settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            elseif ORIGQ then settings().Rendering.QualityLevel = ORIGQ end
        end)
    end
    gui.Destroying:Connect(function() st.boost, st.full = false, false applyLighting() end)
    local function applyHum()
        local hum = getHum()
        if not hum then return end
        pcall(function()
            if st.ws then hum.WalkSpeed = st.ws end
            if st.jp then hum.UseJumpPower = true hum.JumpPower = st.jp end
        end)
    end
    bind(LP.CharacterAdded, function() task.wait(1) applyHum() end)

    W.section(pSet, "เครื่องมือ", 1)
    local afkConn
    W.toggle(pSet, "กันหลุด (Anti-AFK)", "ป้องกันถูกเตะเมื่อไม่ได้ขยับ", false, 2, function(v)
        if afkConn then afkConn:Disconnect() afkConn = nil end
        if v then
            afkConn = LP.Idled:Connect(function()
                pcall(function()
                    local vu = game:GetService("VirtualUser")
                    vu:CaptureController()
                    vu:ClickButton2(Vector2.new())
                end)
            end)
            table.insert(conns, afkConn)
        end
    end)
    W.toggle(pSet, "โหมดเพิ่ม FPS", "ลดคุณภาพกราฟิกและเงา", false, 3, function(v) st.boost = v applyLighting() end)
    W.toggle(pSet, "สว่างเต็มที่ (Fullbright)", nil, false, 4, function(v) st.full = v applyLighting() end)
    W.toggle(pSet, "ปุ่มลัด RightCtrl ซ่อน/แสดงเมนู", nil, true, 5, function(v) hotkeyOn = v end)
    W.section(pSet, "ตัวละคร", 10)
    local initWS, initJP = 16, 50
    local hum0 = getHum()
    if hum0 then
        pcall(function() initWS = math.clamp(math.floor(hum0.WalkSpeed), 1, 200) end)
        pcall(function() initJP = math.clamp(math.floor(hum0.JumpPower), 1, 300) end)
    end
    W.slider(pSet, "ความเร็วเดิน", 1, 200, initWS, 11, function(v) st.ws = v applyHum() end)
    W.slider(pSet, "แรงกระโดด", 1, 300, initJP, 12, function(v) st.jp = v applyHum() end)
    W.section(pSet, "คำสั่งด่วน", 20)
    W.button(pSet, "รีเซ็ตตัวละคร", nil, "ใช้งาน", 21, function() local h = getHum() if h then h.Health = 0 end end)
    W.button(pSet, "เข้าเซิร์ฟใหม่", nil, "ใช้งาน", 22, function()
        pcall(function()
            if #Players:GetPlayers() <= 1 then TeleportService:Teleport(game.PlaceId, LP)
            else TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LP) end
        end)
    end)
    W.button(pSet, "คัดลอก Job ID", nil, "คัดลอก", 23, function()
        if copyText(tostring(game.JobId)) then toast("คัดลอกแล้ว", C.BLUE) else toast("อุปกรณ์นี้ไม่รองรับการคัดลอก", C.RED) end
    end)
    W.button(pSet, "คัดลอก Place ID", nil, "คัดลอก", 24, function()
        if copyText(tostring(game.PlaceId)) then toast("คัดลอกแล้ว ✓", C.BLUE) else toast("อุปกรณ์นี้ไม่รองรับการคัดลอก", C.RED) end
    end)
    W.button(pSet, "ปิดสคริปต์", "หยุดทุกระบบและลบเมนูออก", "ปิด", 25, function() gui:Destroy() end)
