------------------------------------------------------------ helpers
local function new(cls, props, kids)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    for _, c in ipairs(kids or {}) do c.Parent = o end
    return o
end
local function tw(o, t, p, style, dir)
    local tn = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), p)
    tn:Play()
    return tn
end
local function corner(r) return new("UICorner", {CornerRadius = UDim.new(0, r)}) end
local function stroke(col, th, tr)
    return new("UIStroke", {Color = col or C.PURPLE, Thickness = th or 1, Transparency = tr or 0.75, ApplyStrokeMode = Enum.ApplyStrokeMode.Border})
end
local function pad(n)
    return new("UIPadding", {PaddingTop = UDim.new(0, n), PaddingBottom = UDim.new(0, n), PaddingLeft = UDim.new(0, n), PaddingRight = UDim.new(0, n)})
end
local function grad(a, b, rot)
    return new("UIGradient", {Rotation = rot or 0, Color = ColorSequence.new({CK(0, a), CK(1, b)})})
end
local function tri(a, b, c, rot)
    return new("UIGradient", {Rotation = rot or 0, Color = ColorSequence.new({CK(0, a), CK(0.5, b), CK(1, c)})})
end
local function hover(b, over, out)
    b.MouseEnter:Connect(function() tw(b, 0.15, over) end)
    b.MouseLeave:Connect(function() tw(b, 0.2, out) end)
end
local function press(b)
    local s = new("UIScale", {Parent = b})
    b.MouseButton1Down:Connect(function() tw(s, 0.08, {Scale = 0.95}) end)
    local function up() tw(s, 0.25, {Scale = 1}, Enum.EasingStyle.Back) end
    b.MouseButton1Up:Connect(up)
    b.MouseLeave:Connect(up)
end
local function popIn(o, delay)
    local s = o:FindFirstChild("PopScale") or new("UIScale", {Name = "PopScale", Parent = o})
    s.Scale = 0.94
    task.delay(delay or 0, function()
        if s.Parent then tw(s, 0.4, {Scale = 1}, Enum.EasingStyle.Back) end
    end)
end
local function label(parent, props)
    local d = {BackgroundTransparency = 1, Font = Enum.Font.Gotham, TextSize = 14, TextColor3 = C.TEXT, TextXAlignment = Enum.TextXAlignment.Left, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    return new("TextLabel", d)
end
local function button(parent, props, kids)
    local d = {AutoButtonColor = false, Text = "", BorderSizePixel = 0, Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = C.TEXT, Parent = parent}
    for k, v in pairs(props) do d[k] = v end
    local b = new("TextButton", d, kids)
    press(b)
    return b
end
local function glass(parent, props, r)

    return new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(w, h), Position = UDim2.new(0.5, x, 0.5, y), Rotation = rot or 0,

local function copyText(s)
    local fn = (typeof(setclipboard) == "function" and setclipboard)
        or (typeof(toclipboard) == "function" and toclipboard)
        or (typeof(Clipboard) == "table" and Clipboard.set)
    if not fn then return false end
    return (pcall(fn, s))
end
local function fmtTime(s)
    s = math.max(0, math.floor(s))
    return string.format("%02d:%02d:%02d", math.floor(s / 3600), math.floor(s / 60) % 60, s % 60)
end
local function nowStr(fmt)
    local ok, s = pcall(os.date, fmt)
    return ok and s or "--"
end

local function makeDraggable(handle, target)
    local dragging, start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, start, startPos = true, i.Position, target.Position
            target:SetAttribute("moved", false)
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - start
            if d.Magnitude > 5 then target:SetAttribute("moved", true) end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function navIcon(parent, kind)
    local box = new("Frame", {Position = UDim2.fromOffset(10, 8), Size = UDim2.fromOffset(14, 14), BackgroundTransparency = 1, ZIndex = 4, Parent = parent})
    local parts, rings = {}, {}
    local function sq(x, y, w, h, r)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h), BackgroundColor3 = C.MUTED, BorderSizePixel = 0, ZIndex = 4, Parent = box}, {corner(r or 2)})
        table.insert(parts, f)
        return f
    end
    local function ringF(x, y, d, th)
        local f = new("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(d, d), BackgroundTransparency = 1, ZIndex = 4, Parent = box}, {new("UICorner", {CornerRadius = UDim.new(0.5, 0)})})
        local st = new("UIStroke", {Color = C.MUTED, Thickness = th, Parent = f})
        table.insert(rings, st)
        return f
    end
    if kind == "Home" then
        sq(2, 6, 10, 8, 2)
        local rf = sq(2.5, 1, 9, 9, 1)
        rf.Rotation = 45
    elseif kind == "Me" then
        ringF(4, 0, 6, 2)
        sq(1, 8, 12, 6, 3)
    elseif kind == "Utility" then
        sq(0, 0, 6, 6, 2) sq(8, 0, 6, 6, 2) sq(0, 8, 6, 6, 2) sq(8, 8, 6, 6, 2)
    elseif kind == "Farm" then
        ringF(0, 0, 14, 2)
        sq(5, 5, 4, 4, 2)
    elseif kind == "Players" then
        ringF(0, 0, 6, 2) ringF(8, 0, 6, 2)
        sq(0, 8, 6, 6, 3) sq(8, 8, 6, 6, 3)
    elseif kind == "Event" then
        sq(1, 5, 12, 9, 2)
        sq(6, 0, 2, 14, 1)
        sq(0, 3, 14, 3, 1)
    elseif kind == "Egg" then
        sq(2, 0, 10, 14, 5)
    else -- Settings (gear)
        ringF(3, 3, 8, 2)
        sq(6, 0, 2, 3, 1) sq(6, 11, 2, 3, 1) sq(0, 6, 3, 2, 1) sq(11, 6, 3, 2, 1)
        local t1 = sq(1.6, 1.6, 2, 3, 1) t1.Rotation = -45
        local t2 = sq(10.4, 1.6, 2, 3, 1) t2.Rotation = 45
        local t3 = sq(1.6, 9.4, 2, 3, 1) t3.Rotation = 45
        local t4 = sq(10.4, 9.4, 2, 3, 1) t4.Rotation = -45
    end
    return function(col)
        for _, p in ipairs(parts) do p.BackgroundColor3 = col end
        for _, r in ipairs(rings) do r.Color = col end
    end
end

local function setPage(n)

    local hit = new("TextButton", {Text = "", BackgroundTransparency = 1, Position = UDim2.new(0, -8, 0.5, -14), Size = UDim2.new(1, 16, 0, 28), Parent = track})
    local function setRatio(r, fire)
        r = math.clamp(r, 0, 1)
        local v = math.floor(mn + (mx - mn) * r + 0.5)
        fill.Size = UDim2.fromScale(r, 1)
        knob.Position = UDim2.fromScale(r, 0.5)
        val.Text = tostring(v)
        if fire and onChange then onChange(v) end
    end
    setRatio((def - mn) / (mx - mn), false)
    rbtn.MouseButton1Click:Connect(function()
        setRatio((def - mn) / (mx - mn), false)
        if onChange then onChange(def) end
    end)
    local dragging = false
    local function fromInput(i)
        setRatio((i.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), true)
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            list.ScrollingEnabled = false
            fromInput(i)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then fromInput(i) end
    end)
    bind(UIS.InputEnded, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
            dragging = false
            list.ScrollingEnabled = true
        end
    end)

local function flatDist(a, b)
    local dx, dz = a.X - b.X, a.Z - b.Z

local function parseNum(s)
    s = string.lower(tostring(s))
    s = s:gsub("[,%s]", "")
    local num, suf = s:match("^([%d%.]+)(%a*)$")
    local n = tonumber(num)
    if not n then return nil end
    local mult = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }
    if suf ~= "" then n = n * (mult[suf] or 1) end
    return n
end

end

-- one CFrame step toward a goal (flat). freeY = follow the goal height instead of ground height
local function stepTo(hrp, goal, speed, dt, stop, freeY)
    local p = hrp.Position
    local d = flatDist(p, goal)
    if d <= stop then return true end
    local step = math.min(d - stop, speed * dt)
    local nx = p.X + (goal.X - p.X) / d * step
    local nz = p.Z + (goal.Z - p.Z) / d * step
    local y = CFG.GroundY
    if freeY then y = p.Y + (goal.Y + 3 - p.Y) * math.min(1, step / math.max(d, 1)) end
    local np = Vector3.new(nx, y, nz)
    local look = Vector3.new(goal.X, y, goal.Z)
    if (look - np).Magnitude > 0.05 then
        hrp.CFrame = CFrame.lookAt(np, look)
    else
        hrp.CFrame = CFrame.new(np)
    end
    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    return false
end

-- walk with CFrame stepping. true = arrived, false = interrupted / target gone
local function walkTo(getGoal, getSpeed, stop, aliveFn, freeY)
    while aliveFn() do
        local goal = getGoal()
        if not goal then return false end
        local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
        local _, hrp = parts()
        if not hrp then return false end
        if stepTo(hrp, goal, getSpeed(), dt, stop, freeY) then return true end
    end
    return false
end

local function tpTo(pos)

    task.wait(0.15)

    return hrp2 ~= nil and flatDist(hrp2.Position, pos) < 12
end

-- egg rarity: attribute / value -> EggConfig module -> name keyword
local EggCfg
local rarCache = {}
local function configRarity(name)
    if EggCfg == nil then EggCfg = tryRequire({ "Shared", "EggConfig" }) or false end
    if not EggCfg then return nil end
    local function look(t, depth)
        if type(t) ~= "table" or depth > 2 then return nil end
        local e = t[name]
        if type(e) == "table" then
            local r = e.Rarity or e.rarity or e.Tier
            if r then return r end
        end
        for _, v in pairs(t) do
            if type(v) == "table" then
                if v.Name == name or v.EggName == name then
                    local r = v.Rarity or v.rarity
                    if r then return r end
                end
                local r2 = look(v, depth + 1)
                if r2 then return r2 end
            end
        end
        return nil
    end
    local ok, r = pcall(look, EggCfg, 0)
    if ok then return r end
    return nil
end

local function eggRarity(model)
    local key = model.Name
    if rarCache[key] ~= nil then return rarCache[key] or nil end
    local names = { "Rarity", "rarity", "EggRarity", "Tier" }
    local r = readValue(model, names)
    if r == nil then r = readValue(model:FindFirstChild(CFG.EggPart), names) end
    if r == nil then r = configRarity(model.Name) end
    if r == nil then
        local nm = string.lower(model.Name)
        for i = #RARITIES, 1, -1 do
            if nm:find(string.lower(RARITIES[i]), 1, true) then
                r = RARITIES[i]
                break
            end
        end
    end
    local res = false
    if r ~= nil then res = normFrom(RARITIES, r) end
    rarCache[key] = res
    return res or nil
end

-- scan Workspace.Build.ZoneBuilds/<zone>/Eggs/<model>/Egg
local function scanEggs()
    local list, total, unknown = {}, 0, 0
    local o = workspace
    for _, n in ipairs(CFG.ZoneRoot) do o = o and o:FindFirstChild(n) end
    if not o then return list, 0, 0, true end
    local _, hrp = parts()
    for _, zone in ipairs(o:GetChildren()) do
        local eggs = zone:FindFirstChild(CFG.EggsFolder)
        if eggs then
            for _, model in ipairs(eggs:GetChildren()) do
                local part = model:FindFirstChild(CFG.EggPart) or model:FindFirstChildWhichIsA("BasePart", true)
                if part and part:IsA("BasePart") and part.Parent then
                    total = total + 1
                    local r = eggRarity(model)
                    if not r then unknown = unknown + 1 end
                    if r and BR.Rarity[r] then
                        local sz = part.Size
                        if model:IsA("Model") then sz = model:GetExtentsSize() end
                        list[#list + 1] = {
                            part = part, model = model, rarity = r, rank = RANK[r] or 0,
                            size = math.floor(sz.X * sz.Y * sz.Z * 100 + 0.5),
                            half = math.max(part.Size.X, part.Size.Z) / 2,
                            dist = hrp and (part.Position - hrp.Position).Magnitude or 0,
                        }
                    end
                end
            end
        end
    end
    -- bigger egg first, then higher rarity, then nearer
    table.sort(list, function(a, b)
        if a.size ~= b.size then return a.size > b.size end
        if a.rank ~= b.rank then return a.rank > b.rank end
        return a.dist < b.dist
    end)
    return list, total, unknown, false
end

local function sigScore(a, b)
    local n = 0
    for k, v in pairs(a) do
        if b[k] ~= nil and b[k] == v then n = n + 1 end
    end

-- state order: attributes -> BoolValue -> ON/OFF text -> learned look. nil = unknown
local function readSwingState(btn)
    for _, n in ipairs({ "On", "Enabled", "Active", "AutoSwing", "Toggled", "State", "IsOn", "Auto" }) do
        local v = btn:GetAttribute(n)
        if type(v) == "boolean" then return v end
    end
    local bv = btn:FindFirstChildWhichIsA("BoolValue", true)
    if bv then return bv.Value end
    local texts = {}
    pcall(function() texts[#texts + 1] = btn.Text end)
    for _, d in ipairs(btn:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then texts[#texts + 1] = d.Text end
    end
    for _, t in ipairs(texts) do
        local u = string.upper(t)
        if u:find("%f[%a]OFF%f[%A]") then return false end
        if u:find("%f[%a]ON%f[%A]") then return true end
    end
    if Swing.learned then
        local cur = signature(btn)
        local so, sf = sigScore(cur, Swing.learned.on), sigScore(cur, Swing.learned.off)
        if so ~= sf then return so > sf end
    end
    return nil
end

-- make sure auto swing is ON. false = button not found
local function ensureSwing()
    local btn = findSwingBtn()
    if not btn then return false end
    local st = readSwingState(btn)
    if st == true then return true end
    if st == false or (st == nil and not Swing.learned and not Swing.tried) then
        Swing.tried = true
        local before = nil
        if not Swing.learned then before = signature(btn) end -- first press: before = OFF
        fireButton(btn)
        if before then
            task.wait(0.3)
            local after = signature(btn)
            if sigScore(before, after) < countKeys(before) then
                Swing.learned = { off = before, on = after } -- after = ON
            end
        end
    end
    return true
end

local function breakLoop(id)
    local lastEnsure = 0
    local function can() return brAlive(id) and not EventGate.Active end
    while brAlive(id) do
        if EventGate.Active then
            setBrStatus(say("หยุดชั่วคราว (อีเวนท์)", "Paused (event)"))
            task.wait(0.3)
        else
            local list, total, unknown, nofolder = scanEggs()
            if nofolder then
                setBrStatus(say("ไม่พบ Workspace.Build.ZoneBuilds", "Workspace.Build.ZoneBuilds not found"))
                task.wait(1.5)
            elseif #list == 0 then
                setBrStatus(say(
                    "ไม่มีไข่ที่ตรงเงื่อนไข (พบ " .. total .. " ใบ, ไม่รู้ความหายาก " .. unknown .. ") รอสแกนใหม่",
                    "No matching egg (" .. total .. " found, " .. unknown .. " unknown rarity), rescanning"))
                task.wait(1.2)
            else
                local egg = list[1]
                local nm = egg.model.Name .. " [" .. egg.rarity .. "]"
                local stop = math.max(CFG.EggStop, egg.half + CFG.EggGap)
                setBrStatus(say("เดินไปหาไข่ ", "Walking to egg ") .. nm)
                local arrived = walkTo(function()
                    if egg.part.Parent then return egg.part.Position end
                    return nil
                end, function() return BR.Speed end, stop, can)
                if arrived then
                    setBrStatus(say("กำลังทุบไข่ ", "Breaking egg ") .. nm)
                    local t0 = os.clock()
                    while can() and egg.part.Parent and os.clock() - t0 < 45 do
                        local _, hrp = parts()
                        if not hrp then break end
                        local gp = egg.part.Position
                        if flatDist(hrp.Position, gp) > stop + 2 then break end
                        hrp.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(gp.X, hrp.Position.Y, gp.Z))
                        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                        if os.clock() - lastEnsure > 0.6 then
                            lastEnsure = os.clock()
                            ensurePickaxe()
                            if not ensureSwing() then
                                setBrStatus(say("ไม่พบปุ่ม AutoSwing → ใช้การตีแทน", "AutoSwing button not found -> fallback swing"))
                                fallbackSwing()
                            end
                        end
                        task.wait(0.1)
                    end
                end
            end
        end
    end
end

local function stopBreak()
    BR.Running = false
    BR.Job = BR.Job + 1
    setBrStatus(say("หยุดทำงาน", "Stopped"))
end

local function startBreak()
    local any = false
    for _ in pairs(BR.Rarity) do any = true break end
    if not any then
        toast(T("n_rarity"), C.RED)
        return false
    end
    BR.Job = BR.Job + 1
    BR.Running = true
    local id = BR.Job
    task.spawn(function()
        local ok, err = pcall(breakLoop, id)
        if not ok then
            warn("[NEXORA Auto Break]", err)
            if BR.Job == id then
                BR.Running = false
                BR.Job = BR.Job + 1
                setBrStatus(say("ผิดพลาด: ", "Error: ") .. tostring(err))
                brSet(false, false) -- reset the UI toggle
            end
        end
    end)
    return true
end

    local _, _, hum = parts()
    if hum then
        lockHum = { ws = hum.WalkSpeed, jp = hum.JumpPower, ujp = hum.UseJumpPower }
        hum.WalkSpeed = 0
        hum.UseJumpPower = true
        hum.JumpPower = 0
        if ST.AntiRagdoll then
            pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false) end)
            pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false) end)
        end
    end
    busyConn = RunService.Stepped:Connect(function()
        local ch = LP.Character
        if not ch then return end
        if ST.Noclip then
            for _, d in ipairs(ch:GetDescendants()) do
                if d:IsA("BasePart") then
                    if noclipOrig[d] == nil then noclipOrig[d] = d.CanCollide end
                    d.CanCollide = false
                end
            end
        end
        if ST.AntiRagdoll then
            for _, d in ipairs(ch:GetDescendants()) do
                if d:IsA("Motor6D") then
                    d.Enabled = true
                elseif d:IsA("BallSocketConstraint") then
                    if ballOrig[d] == nil then ballOrig[d] = d.Enabled end
                    d.Enabled = false
                end
            end
        end
    end)
end

local function lockOff()
    if not ST.Busy then return end
    ST.Busy = false
    EventGate.StealBusy = false
    if busyConn then
        busyConn:Disconnect()
        busyConn = nil
    end
    pcall(function() CAS:UnbindAction("NXStealLock") end)
    local _, _, hum = parts()
    if hum and lockHum.ws then
        hum.WalkSpeed = lockHum.ws
        hum.UseJumpPower = lockHum.ujp
        hum.JumpPower = lockHum.jp
    end
    if hum then
        pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true) end)
        pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true) end)
    end
    lockHum = {}
    for part, v in pairs(noclipOrig) do
        if part.Parent then part.CanCollide = v end
        noclipOrig[part] = nil
    end
    for c, v in pairs(ballOrig) do
        if c.Parent then c.Enabled = v end
        ballOrig[c] = nil
    end
end

-- animal info (attributes / values on the pickup model)
local function animalInfo(m)
    local r = readValue(m, { "Rarity", "rarity" })
    local mu = readValue(m, { "Mutation", "mutation" })
    local inc = readValue(m, { "IncomePerSecond", "Income", "IncomePerSec", "CashPerSecond", "PerSecond" })
    local info = {}
    if r ~= nil then info.rarity = normFrom(RARITIES, r) end
    if mu ~= nil then
        local s = tostring(mu)
        if s == "" or string.lower(s) == "none" then s = "Normal" end
        info.mutation = normFrom(MUTATIONS, s)
    end
    info.income = tonumber(inc)
    return info
end

-- best animal that passes the rarity / mutation / min-income filters
local function pickBest(pickups)
    local _, hrp = parts()
    local best
    for _, m in ipairs(pickups:GetChildren()) do
        local pos = posOf(m)
        if pos then
            local info = animalInfo(m)
            local rOk = next(ST.Rarity) == nil or info.rarity == nil or ST.Rarity[info.rarity] == true
            local mOk = next(ST.Mutation) == nil or ST.Mutation[info.mutation or "Normal"] == true
            local iOk = ST.MinIncome <= 0 or info.income == nil or info.income >= ST.MinIncome
            if rOk and mOk and iOk then
                local c = {
                    model = m, rank = RANK[info.rarity or ""] or 0, income = info.income or 0,
                    dist = hrp and (pos - hrp.Position).Magnitude or 0,
                }
                if not best or c.rank > best.rank
                    or (c.rank == best.rank and (c.income > best.income
                        or (c.income == best.income and c.dist < best.dist))) then
                    best = c
                end
            end
        end
    end
    return best and best.model or nil
end

-- go to a position: teleport first in TP mode (retry, then fall back to walking)
local function travel(getPos, stop, ok)
    if ST.TpMode then
        for _ = 1, 3 do
            if not ok() then return false end
            local pos = getPos()
            if not pos then return false end
            if tpTo(pos) then return true end
            stSay("วาร์ปไม่สำเร็จ → เดินแทน", "Teleport failed -> walking")
            task.wait(0.2)
        end
    end
    return walkTo(getPos, function() return ST.Speed end, stop, ok)
end

local function goSafe(id)
    local sz = findNamed("safe", CFG.SafeNames)
    local pos = sz and posOf(sz)
    if not pos then return false end
    local _, hrp = parts()
    if not hrp then return false end
    if flatDist(hrp.Position, pos) < 6 then return true end
    return travel(function() return pos end, 3, function() return stAlive(id) and not EventGate.Active end)
end

-- guard: walk to it and wait until we get hit (health drops) or time runs out
local function handleGuard(ok)
    local g = findNamed("guard", CFG.GuardNames)
    local gp = g and posOf(g)
    if not gp then return end
    stSay("เดินไปหายาม", "Walking to the guard")
    walkTo(function() return gp end, function() return ST.Speed end, 4, ok)
    local _, _, hum = parts()
    local h0 = hum and hum.Health or 0
    local t0 = os.clock()
    while ok() and os.clock() - t0 < 3 do
        local _, _, hm = parts()
        if hm and hm.Health < h0 then break end
        task.wait(0.1)
    end
end

-- deliver to START (or the original spot). If the animal is dropped, pick it up again.
local function deliver(origin, pickups, target, ok)
    local dest = findNamed("start", CFG.StartNames)
    local dpos = (dest and posOf(dest)) or origin
    if not dpos then return end
    for _ = 1, 3 do
        stSay("ไปส่งที่ START", "Delivering to START")
        local dropped = false
        travel(function() return dpos end, 3, function()
            if target.Parent == pickups then
                dropped = true
                return false
            end
            return ok()
        end)
        if not dropped then return end
        stSay("สัตว์หล่น → เก็บใหม่", "Dropped -> picking it up again")
        walkTo(function() return posOf(target) end, function() return ST.Speed end, CFG.StealStop, ok)
        local p = findPrompt(target)
        if p then
            firePrompt(p)
            task.wait(0.5)
        end
    end
end

-- TP MODE pre-phase: smallest egg in the current zone -> break -> pick the dropped animal -> guard
local function currentZoneEggs()
    local o = workspace
    for _, n in ipairs(CFG.ZoneRoot) do o = o and o:FindFirstChild(n) end
    if not o then return nil end
    local _, hrp = parts()
    if not hrp then return nil end
    local bestF, bd
    for _, zone in ipairs(o:GetChildren()) do
        local eggs = zone:FindFirstChild(CFG.EggsFolder)
        local first = eggs and eggs:GetChildren()[1]
        local p = first and posOf(first)
        if p then
            local d = flatDist(p, hrp.Position)
            if not bd or d < bd then bestF, bd = eggs, d end
        end
    end
    return bestF
end

local function smallestEgg(folder)
    local _, hrp = parts()
    local best
    for _, m in ipairs(folder:GetChildren()) do
        local part = m:FindFirstChild(CFG.EggPart) or m:FindFirstChildWhichIsA("BasePart", true)
        if part and part:IsA("BasePart") and part.Parent then
            local sz = part.Size
            if m:IsA("Model") then sz = m:GetExtentsSize() end
            local c = {
                part = part, size = math.floor(sz.X * sz.Y * sz.Z * 100 + 0.5),
                half = math.max(part.Size.X, part.Size.Z) / 2,
                dist = hrp and (part.Position - hrp.Position).Magnitude or 0,
            }
            if not best or c.size < best.size or (c.size == best.size and c.dist < best.dist) then best = c end
        end
    end
    return best
end

local function tpPhase(pickups, ok)
    local folder = currentZoneEggs()
    local egg = folder and smallestEgg(folder)
    if egg then
        stSay("วาร์ป: เดินไปหาไข่", "Teleport: walking to an egg")
        local stop = math.max(CFG.EggStop, egg.half + CFG.EggGap)
        local arrived = walkTo(function()
            if egg.part.Parent then return egg.part.Position end
            return nil
        end, function() return ST.Speed end, stop, ok)
        if arrived then
            stSay("วาร์ป: กำลังทุบไข่", "Teleport: breaking the egg")
            local epos = egg.part.Position
            local t0 = os.clock()
            while ok() and egg.part.Parent and os.clock() - t0 < 45 do
                ensurePickaxe()
                if not ensureSwing() then fallbackSwing() end
                task.wait(0.25)
            end
            local drop
            local t1 = os.clock()
            while ok() and not drop and os.clock() - t1 < 4 do
                local bd
                for _, m in ipairs(pickups:GetChildren()) do
                    local p = posOf(m)
                    if p and (p - epos).Magnitude < 40 then
                        local d = (p - epos).Magnitude
                        if not bd or d < bd then drop, bd = m, d end
                    end
                end
                if not drop then task.wait(0.2) end
            end
            if drop and ok() then
                stSay("วาร์ป: เก็บสัตว์ที่หล่น", "Teleport: picking up the dropped animal")
                walkTo(function() return posOf(drop) end, function() return ST.Speed end, CFG.StealStop, ok)
                local p = findPrompt(drop)
                if p then
                    firePrompt(p)
                    task.wait(0.5)
                end
            end
        end
    end
    if ok() then handleGuard(ok) end
end

-- one steal run
local function runSteal(id, pickups, target, origin)
    local function ok() return stAlive(id) and not EventGate.Active end
    lockOn()
    local okp, err = pcall(function()
        local nm = target.Name
        if ST.TpMode then tpPhase(pickups, ok) end
        if not ok() or not target.Parent then return end
        stSay("เดินไปหา " .. nm, "Walking to " .. nm)
        if not travel(function() return posOf(target) end, CFG.StealStop, ok) then return end
        local picked = false
        for _ = 1, 4 do
            if not ok() then break end
            local p = findPrompt(target)
            if not p then
                stSay("ไม่พบพรอมต์ขโมย", "Steal prompt not found")
                task.wait(0.6)
            else
                stSay("กำลังขโมย " .. nm, "Stealing " .. nm)
                firePrompt(p)
                task.wait(0.5)
                local ch = LP.Character
                if target.Parent ~= pickups or (ch and target:IsDescendantOf(ch)) then
                    picked = true
                    break
                end
            end
        end
        if not picked then return end -- lost -> next target
        ST.Collected = ST.Collected + 1
        if not ST.TpMode then handleGuard(ok) end
        deliver(origin, pickups, target, ok)
        stSay("ส่งเสร็จ", "Delivered")
    end)
    lockOff() -- never leave controls locked, even after an error
    if not okp then error(err, 0) end
end

local function stealLoop(id)
    local _, hrp0 = parts()
    local origin = hrp0 and hrp0.Position or nil
    local pickups
    while stAlive(id) and not pickups do
        pickups = workspace:FindFirstChild(CFG.PickupsFolder, true)
        if not pickups then
            stSay("รอโฟลเดอร์ AnimalPickups", "Waiting for AnimalPickups")
            task.wait(1)
        end
    end
    while stAlive(id) do
        if not pickups.Parent then
            pickups = workspace:FindFirstChild(CFG.PickupsFolder, true) or pickups
        end
        if EventGate.Active then
            stSay("หยุดชั่วคราว (อีเวนท์)", "Paused (event)")
            task.wait(0.3)
        else
            local target = pickBest(pickups)
            if not target then
                stSay("ไม่มีเป้าหมาย → รอในโซนปลอดภัย", "No target -> waiting in the Safe Zone")
                goSafe(id)
                task.wait(1)
            else
                if ST.SafeWait then
                    stSay("รอในโซนปลอดภัยก่อนรัน", "Waiting in the Safe Zone")
                    goSafe(id)
                    task.wait(0.8)
                    target = stAlive(id) and pickBest(pickups) or nil
                end
                if target and stAlive(id) and not EventGate.Active then
                    runSteal(id, pickups, target, origin)
                    -- return to the Safe Zone, or the original spot
                    if stAlive(id) and not EventGate.Active then
                        if not goSafe(id) and origin then
                            travel(function() return origin end, 3, function() return stAlive(id) and not EventGate.Active end)
                        end
                    end
                end
            end
        end
    end
end

local function stopSteal()
    ST.Running = false
    ST.Job = ST.Job + 1
    lockOff()
    stSay("หยุดทำงาน", "Stopped")
end

local function startSteal()
    ST.Job = ST.Job + 1
    ST.Running = true
    local id = ST.Job
    task.spawn(function()
        local ok, err = pcall(stealLoop, id)
        if not ok then
            warn("[NEXORA Auto Steal]", err)
            lockOff()
            if ST.Job == id then
                ST.Running = false
                ST.Job = ST.Job + 1
                stSay("ผิดพลาด: " .. tostring(err), "Error: " .. tostring(err))
                stSet(false, false) -- reset the UI toggle
            end
        end
    end)
    return true
end

    local g = { items = setmetatable({}, { __mode = "k" }), on = false }

local function consider(d)
    if not d:IsA("BasePart") then return end
    local hasTrigger = d:FindFirstChildOfClass("ProximityPrompt") or d:FindFirstChildOfClass("TouchTransmitter")
    if not hasTrigger then return end
    if d:FindFirstAncestor(CFG.PickupsFolder) or d:FindFirstAncestor(CFG.ZoneRoot[2]) then return end
    local m = d.Parent
    if Players:GetPlayerFromCharacter(d) or (m and Players:GetPlayerFromCharacter(m)) then return end
    local name
    if evMatch(d.Name) then
        name = d.Name
    elseif m and m ~= workspace and evMatch(m.Name) then
        name = m.Name
    end
    if not name then return end
    if not groups[name] then addGroup(name) end
    groups[name].items[d] = true
end

-- collector: pauses Auto Break / Auto Steal through EventGate, resumes afterwards
task.spawn(function()
    local done = setmetatable({}, { __mode = "k" })

        local anyOn = false
        if not EventGate.StealBusy then
            local _, hrp = parts()

                if g.on then

                    if hrp then
                        for p in pairs(g.items) do
                            local doneAt = done[p]
                            if p.Parent and not (doneAt and os.clock() - doneAt < 20) then
                                local d = (p.Position - hrp.Position).Magnitude
                                if not bd or d < bd then best, bestG, bd = p, g, d end
                            end
                        end
                    end
                end
            end
        end
        if not best then
            Hold.collecting = false
            refreshGate()
            task.wait(0.6)
        else
            Hold.collecting = true
            refreshGate()
            walkTo(function()
                if best.Parent then return best.Position end
                return nil
            end, function() return 120 end, 2, function()
                return gui.Parent ~= nil and bestG.on and not EventGate.StealBusy
            end, true)
            local pr = best:FindFirstChildOfClass("ProximityPrompt")
            local _, hrp2 = parts()
            if pr then
                firePrompt(pr)
            elseif hrp2 and typeof(firetouchinterest) == "function" then
                pcall(firetouchinterest, hrp2, best, 0)

------------------------------------------------------------ live loop (FPS / ping / health / position)
local frames, acc = 0, 0
bind(RunService.Heartbeat, function(dt)
    frames = frames + 1
    acc = acc + dt
end)
local function getPing()
    local p
    pcall(function() p = game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() end)
    if not p then pcall(function() p = LP:GetNetworkPing() * 1000 end) end
    return p or 0
end
local startT = os.clock()
local fpsNow = 0
task.spawn(function()
    local n = 0
    while gui.Parent do
        task.wait(0.25)
        n = n + 1
        local inst = (acc > 0) and (frames / acc) or 0
        frames, acc = 0, 0
        fpsNow = (fpsNow == 0) and inst or (fpsNow * 0.4 + inst * 0.6)
        local fps = math.floor(fpsNow + 0.5)
        local ping = math.floor(getPing() + 0.5)
        local hum, hrp = getHum()
        local hp = hum and hum.Health or 0
        local mhp = hum and hum.MaxHealth or 100
        if mhp <= 0 then mhp = 100 end
        local pct = math.clamp(hp / mhp, 0, 1)
        ringFill = math.floor(pct * RING_N + 0.5)

        if n % 4 == 0 then -- push one sample / second into the chart
            table.remove(hist, 1)
            table.insert(hist, fps)
            if holder.Visible and home.Visible then drawChart() end
        end

        if holder.Visible then
            local sess = fmtTime(os.clock() - startT)
            heroClock.Text = nowStr("%H:%M:%S")
            heroSub.Text = nowStr("%d/%m/%Y") .. "  ·  " .. sess
            if home.Visible then
                stPlayers.Text = tostring(#Players:GetPlayers())
                stFps.Text = tostring(fps)
                stPing.Text = ping .. " ms"
                fpsTop.Text = tostring(fps)
                ringPct.Text = math.floor(pct * 100 + 0.5) .. "%"
                legendVals[1].Text = math.floor(hp) .. "/" .. math.floor(mhp)
                legendVals[2].Text = tostring(fps)
                legendVals[3].Text = ping .. "ms"
            end
            if pages.Me.Visible then
                hpText.Text = math.floor(hp) .. " / " .. math.floor(mhp)
                tw(hpFill, 0.25, {Size = UDim2.fromScale(pct, 1)})
                tFps.Text = tostring(fps)
                tPing.Text = ping .. " ms"
                tSpeed.Text = hum and tostring(math.floor(hum.WalkSpeed)) or "-"
                tSess.Text = sess
                if hrp then
                    local p = hrp.Position
                    infoVals.position.Text = math.floor(p.X) .. ", " .. math.floor(p.Y) .. ", " .. math.floor(p.Z)
                else
                    infoVals.position.Text = "-"
                end
                infoVals.in_server.Text = #Players:GetPlayers() .. " / " .. Players.MaxPlayers
                infoVals.acc_age.Text = LP.AccountAge .. " " .. T("days")
                infoVals.server.Text = (game.JobId ~= "" and string.sub(game.JobId, 1, 8)) or "Studio"
            end
        end
    end
end)
