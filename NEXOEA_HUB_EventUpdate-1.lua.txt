local function compile(source, label)
	local fn, err = loadstring(source, label)
	if not fn then
		return nil, err
	end
	local ok, result = pcall(fn)
	if not ok then
		return nil, result
	end
	return result
end

local function fetchLibrary()

	if type(rawget(ENV, "LunarLibrary")) == "table" and type(ENV.LunarLibrary.new) == "function" then
		return ENV.LunarLibrary
	end

	if type(CONFIG.LibraryUrl) == "string" and CONFIG.LibraryUrl ~= "" and not CONFIG.LibraryUrl:find("YOUR_NAME", 1, true) then
		for attempt = 1, CONFIG.Retries do
			local ok, source = pcall(function()
				return game:HttpGet(CONFIG.LibraryUrl)
			end)
			if ok and type(source) == "string" and #source > 0 then
				local lib, err = compile(source, "=LUNAR_Library")
				if lib then
					return lib
				end
				warn("[NEXOEA Loader] Library compile error: " .. tostring(err))
				break
			end
			log(("download failed (%d/%d)"):format(attempt, CONFIG.Retries))
			task.wait(CONFIG.RetryDelay)
		end
	end

	if type(isfile) == "function" and type(readfile) == "function" and isfile(CONFIG.LibraryFile) then
		local lib, err = compile(readfile(CONFIG.LibraryFile), "=LUNAR_Library")
		if lib then
			return lib
		end
		warn("[NEXOEA Loader] Local library error: " .. tostring(err))
	end

	return nil
end

function Stats.Steal(muts, sign)
	Stats.Stolen = math.max(0, Stats.Stolen + sign)
	if muts and next(muts) ~= nil then
		Stats.MutStolen = math.max(0, Stats.MutStolen + sign)
		for name in pairs(muts) do
			Stats.Mut[name] = math.max(0, (Stats.Mut[name] or 0) + sign)
		end
	end
end

local function richEsc(v)
	return (tostring(v):gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function hex3(c)
	return string.format("#%02X%02X%02X", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
end

function Stats.RarityColor3(name)
	local st = name and RARITY_STYLE[name]
	local ok, c = pcall(Color3.fromHex, st and st[1] or "#9AA0A6")
	return ok and c or Color3.fromRGB(154, 160, 166)
end

function Stats.RarityRich(name, text, bold)
	text = tostring(text or name or "?")
	local st = name and RARITY_STYLE[name]
	local out
	if st and st.rainbow then
		local chars = {}
		local okc = pcall(function()
			for _, cp in utf8.codes(text) do
				chars[#chars + 1] = utf8.char(cp)
			end
		end)
		if not okc or #chars == 0 then
			chars = {}
			for i = 1, #text do
				chars[#chars + 1] = text:sub(i, i)
			end
		end
		local n = math.max(#chars, 1)
		local parts = {}
		for i, ch in ipairs(chars) do
			local c = Color3.fromHSV(((i - 1) / n) * 0.85, 0.8, 1)
			parts[#parts + 1] = ('<font color="%s">%s</font>'):format(hex3(c), richEsc(ch))
		end
		out = table.concat(parts)
	else
		out = ('<font color="%s">%s</font>'):format(st and st[1] or "#9AA0A6", richEsc(text))
		if st and st.stroke then
			out = ('<stroke color="%s" joins="miter" thickness="1" transparency="0.3">%s</stroke>'):format(st.stroke, out)
		end
	end
	if bold then
		out = "<b>" .. out .. "</b>"
	end
	return out
end

function Stats.ViewClone(inst)
	if typeof(inst) ~= "Instance" then
		return nil
	end
	-- Some game models are Archivable = false (or have non-archivable children). Clone() then
	-- returns nil or skips them, which is what caused the "?" / empty icons. Unlock, clone, restore.
	local flipped = {}
	local function unlock(i)
		local okA, a = pcall(function()
			return i.Archivable
		end)
		if okA and a == false then
			if pcall(function()
				i.Archivable = true
			end) then
				flipped[#flipped + 1] = i
			end
		end
	end
	unlock(inst)
	local okD, desc = pcall(function()
		return inst:GetDescendants()
	end)
	if okD then
		for _, d in ipairs(desc) do
			unlock(d)
		end
	end
	local ok, c = pcall(function()
		return inst:Clone()
	end)
	for _, i in ipairs(flipped) do
		pcall(function()
			i.Archivable = false
		end)
	end
	if not ok or not c then
		return nil
	end
	if c:IsA("LuaSourceContainer") or c:IsA("GuiBase2d") then
		pcall(function()
			c:Destroy()
		end)
		return nil
	end
	local kill = {}
	for _, d in ipairs(c:GetDescendants()) do
		if d:IsA("LuaSourceContainer") or d:IsA("GuiBase2d") or d:IsA("Sound") or d:IsA("ParticleEmitter")
			or d:IsA("Trail") or d:IsA("Beam") or d:IsA("Light") then
			kill[#kill + 1] = d
		end
	end
	for _, d in ipairs(kill) do
		pcall(function()
			d:Destroy()
		end)
	end
	local parts = {}
	if c:IsA("BasePart") then
		parts[1] = c
	end
	for _, d in ipairs(c:GetDescendants()) do
		if d:IsA("BasePart") then
			parts[#parts + 1] = d
		end
	end
	if #parts == 0 then
		pcall(function()
			c:Destroy()
		end)
		return nil
	end
	local visible = 0
	for _, pt in ipairs(parts) do
		if pt.Transparency < 0.95 then
			visible += 1
		end
	end
	for _, pt in ipairs(parts) do
		pt.Anchored = true
		-- invisible hitboxes make the camera frame far too wide (tiny / empty icon)
		if visible > 0 and pt ~= c and pt.Transparency >= 0.95 then
			pcall(function()
				pt:Destroy()
			end)
		end
	end
	return c
end

function Stats.NoteEggSize(sz)
	if type(sz) == "number" and sz > Stats.MaxEggSize then
		Stats.MaxEggSize = sz
	end
end

function Stats.SizeScore(raw)
	if type(raw) ~= "number" or raw <= 0 or Stats.MaxEggSize <= 0 then
		return nil
	end
	return math.clamp(math.floor(raw / Stats.MaxEggSize * 100 + 0.5), 1, 100)
end

function Stats.AddSteal(muts, info)
	if not Stats.On then
		return nil
	end
	info = info or {}
	muts = muts or {}
	Stats.Steal(muts, 1)
	Stats.NextId += 1
	local names = {}
	for n in pairs(muts) do
		names[#names + 1] = n
	end
	table.sort(names)
	local entry = {
		Id = Stats.NextId,
		Kind = "Steal",
		At = os.time(),
		AfkAt = select(2, pcall(Stats.AfkClock)),
		Name = info.Name or "?",
		Rarity = info.Rarity,
		Income = info.Income,
		Muts = names,
		MutSet = muts,
		Model = info.Model,
	}
	table.insert(Stats.History.Steal, 1, entry)
	histTrim(Stats.History.Steal)
	Stats.History.Rev += 1
	return entry
end

function Stats.UndoSteal(entry)
	if not entry then
		return
	end
	Stats.Steal(entry.MutSet or {}, -1)
	local list = Stats.History.Steal
	local idx = table.find(list, entry)
	if idx then
		table.remove(list, idx)
	end
	histDrop(entry)
	Stats.History.Rev += 1
end

function Stats.AddEgg(info)
	if not Stats.On then
		return
	end
	info = info or {}
	Stats.Eggs += 1
	Stats.NoteEggSize(info.Size)
	Stats.NextId += 1
	local entry = {
		Id = Stats.NextId,
		Kind = "Egg",
		At = os.time(),
		AfkAt = select(2, pcall(Stats.AfkClock)),
		Name = info.Name or "?",
		Rarity = info.Rarity,
		MaxHealth = info.MaxHealth,
		Size = info.Size,
		Model = info.Model,
	}
	table.insert(Stats.History.Eggs, 1, entry)
	histTrim(Stats.History.Eggs)
	Stats.History.Rev += 1
end

	local function wordState(t)
		t = t:lower()
		if t:find("%f[%a]off%f[%A]") then
			return false
		end
		if t:find("%f[%a]on%f[%A]") then
			return true
		end
		return nil
	end

	local swingHooked = {}
	local function setSwingEventState(want)
		if type(getnilinstances) ~= "function" then
			return false
		end
		local ok, list = pcall(getnilinstances)
		if not ok or type(list) ~= "table" then
			return false
		end
		for _, obj in ipairs(list) do
			local matches = false
			pcall(function()
				matches = obj.Name == "Event"
					and obj:IsA("BindableEvent")
					and obj:GetDebugId() == "0_132375"
			end)
			if matches then
				local fired = pcall(function()
					obj:Fire(want)
				end)
				if fired then
					SW.EvtState = want
					return true
				end
			end
		end
		return false
	end
	local function hookSwingEvents()
		if type(getnilinstances) ~= "function" then
			return
		end
		local ok, list = pcall(getnilinstances)
		if not ok or type(list) ~= "table" then
			return
		end
		for _, o in ipairs(list) do
			local isEvt = false
			pcall(function()
				isEvt = o.Name == "Event" and o:IsA("BindableEvent")
			end)
			if isEvt and not swingHooked[o] then
				swingHooked[o] = true
				pcall(function()
					track(o.Event:Connect(function(v)
						if type(v) == "boolean" then
							SW.EvtState = v
						end
					end))
				end)
			end
		end
	end

	local function readSwingState(btn)
		for _, inst in ipairs({ btn, btn.Parent }) do
			if inst then
				for k, v in pairs(inst:GetAttributes()) do
					if type(v) == "boolean" then
						local n = k:lower()
						if n:find("auto") or n:find("enabled") or n:find("active") or n == "on" or n:find("toggle") or n:find("state") then
							return v
						end
					end
				end
			end
		end
		for _, d in ipairs(btn:GetChildren()) do
			if d:IsA("BoolValue") then
				return d.Value
			end
		end
		local nodes = btn:GetDescendants()
		table.insert(nodes, 1, btn)
		for _, d in ipairs(nodes) do
			if d:IsA("TextLabel") or d:IsA("TextButton") then
				local r = wordState(d.Text)
				if r ~= nil then
					return r
				end
			end
		end
		if SW.EvtState ~= nil then
			return SW.EvtState
		end
		if SW.OnSig and SW.OffSig then
			local s = sigOf(btn)
			if s == SW.OnSig then
				return true
			elseif s == SW.OffSig then
				return false
			end
		end
		return nil
	end

	end

	local function enableGameSwing()
		hookSwingEvents()
		local btn = ensureSwingRemote and ensureSwingRemote() or findSwingBtn()
		if not btn then

			setSwingEventState(true)
			Notify("Game AutoSwing button not found (walking only)", false)
			return
		end

		SW.EvtState = nil
		SW.Assumed = nil
		local current = readSwingState(btn)
		local before = sigOf(btn)

		if current ~= true then
			if not pressBtn(btn) then
				Notify("firesignal not available - cannot press AutoSwing", false)
				setSwingEventState(true)
				return
			end
			task.wait(0.35)
		end

		setSwingEventState(true)
		SW.Assumed = true

		for _ = 1, 2 do
			local st = readSwingState(btn)
			if st ~= false then
				break
			end
			pressBtn(btn)
			task.wait(0.35)
		end

		local after = sigOf(btn)
		if before ~= after and readSwingState(btn) ~= false then
			SW.OffSig, SW.OnSig = before, after
		end
	end

	local function monitorSwing(id)
		local lastPress = 0
		local lastHook = 0
		local wasHolding = nil
		while swAlive(id) do
			task.wait(0.15)
			if os.clock() - lastHook > 5 then
				lastHook = os.clock()
				hookSwingEvents()
			end
			if swAlive(id) and not SW.Yielded and not SW.Resuming and not EventGate.Active then

				local holding = holdingPick()
				if wasHolding == false and holding then
					enableGameSwing()
					setSwStatus("Pickaxe back in hand -> AutoSwing turned ON again")
				end
				wasHolding = holding
				if SW.EvtState == false then
					setSwingEventState(true)
				end
				local btn = findSwingBtn()
				if not btnShown(btn) and ensureSwingRemote then
					btn = ensureSwingRemote()
				end

				if btn and readSwingState(btn) == false and os.clock() - lastPress > 0.6 then
					lastPress = os.clock()
					pressBtn(btn)
					SW.Assumed = true
					setSwStatus("AutoSwing was OFF -> turned ON again")
				end
			else
				wasHolding = nil
			end
		end
	end

	local function isPickaxe(t)
		return t:IsA("Tool") and t.Name:lower():find("pick") ~= nil
	end

	local function equipPickaxe()
		local char = LocalPlayer.Character
		local hum = getHumanoid()
		if not (char and hum) then
			return false
		end
		for _, t in ipairs(char:GetChildren()) do
			if isPickaxe(t) then
				return true
			end
		end
		local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
		if not bp then
			return false
		end
		for _, t in ipairs(bp:GetChildren()) do
			if isPickaxe(t) then
				pcall(function()
					hum:EquipTool(t)
				end)
				return true
			end
		end
		return false
	end

	local function reequipPickaxe()
		local hum = getHumanoid()
		if hum then
			pcall(function()
				hum:UnequipTools()
			end)
		end
		task.wait(0.3)
		equipPickaxe()
	end

	ensureSwingRemote = function()
		local btn = findSwingBtn()
		for _ = 1, 3 do
			if btnShown(btn) then
				return btn
			end
			if setSwStatus then
				setSwStatus("Swing button not showing -> re-equipping pickaxe")
			end
			reequipPickaxe()
			task.wait(0.6)
			btn = findSwingBtn()
		end
		return btn
	end

	local function eggRarity(model)
		local g = model:FindFirstChild("RarityGradient", true)
		return g and g:GetAttribute("Rarity")
	end

	local function eggAlive(part)
		return part and part.Parent and part:GetAttribute("Broken") ~= true and (part:GetAttribute("Health") or 1) > 0
	end

	local function eggSize(model, part)
		local ok, s = pcall(function()
			return model:GetExtentsSize()
		end)
		if not ok or not s then
			s = part.Size
		end
		return math.floor(s.X * s.Y * s.Z * 100 + 0.5)
	end

	local function scanEggs()
		local out = {}
		local build = Workspace:FindFirstChild("Build")
		local zb = build and build:FindFirstChild("ZoneBuilds")
		if not zb then
			return out
		end
		local hrp = getHRP()
		for _, zone in ipairs(zb:GetChildren()) do
			local eggs = zone:FindFirstChild("Eggs")
			if eggs then
				for _, m in ipairs(eggs:GetChildren()) do
					local part = m:FindFirstChild("Egg")
					if part and part:IsA("BasePart") and eggAlive(part) then
						local r = eggRarity(m)
						local wanted = r and SW.Rarity[r]
						local sz
						if wanted then
							sz = eggSize(m, part)
						else
							sz = Stats.EggSizeSeen[m]
							if sz == nil then
								sz = eggSize(m, part)
								Stats.EggSizeSeen[m] = sz
							end
						end
						Stats.NoteEggSize(sz)
						if wanted then
							table.insert(out, {
								part = part,
								model = m,
								name = m.Name,
								rarityName = r,
								size = sz,
								rank = RANK[r] or 0,
								dist = hrp and (part.Position - hrp.Position).Magnitude or 0,
							})
						end
					end
				end
			end
		end

		table.sort(out, function(a, b)
			if a.size ~= b.size then
				return a.size > b.size
			end
			if a.rank ~= b.rank then
				return a.rank > b.rank
			end
			return a.dist < b.dist
		end)
		return out
	end

	function Stats.SnapEgg(e)
		if not Stats.On or not e or not e.part then
			return nil
		end
		local part = e.part
		local cached = Stats.EggSnapCache[part]
		if cached then
			return cached
		end
		local model = e.model or part.Parent
		local hp = tonumber(part:GetAttribute("MaxHealth")) or tonumber(part:GetAttribute("Health"))
		local rar = e.rarityName
		if not rar and model then
			rar = eggRarity(model)
		end
		local snap = {
			Name = tostring(e.name or (model and model.Name) or part.Name),
			Rarity = rar,
			MaxHealth = hp,
			Size = e.size or eggSize(model, part),
			Model = Stats.ViewClone(model or part),
		}
		Stats.EggSnapCache[part] = snap
		return snap
	end

	local function swWalkTo(part, id)
		if SW.TpMode then
			local done = TPX.reach({
				part = part,
				stop = SW_STOP,
				say = setSwStatus,
				alive = function()
					return swAlive(id) and not breakBlocked() and eggAlive(part)
				end,
			})
			if done then
				return true
			end
		end
		while swAlive(id) and not breakBlocked() and eggAlive(part) do
			local r = getHRP()
			if not r then
				return false
			end
			local dir = Vector3.new(part.Position.X - r.Position.X, 0, part.Position.Z - r.Position.Z)
			local dist = dir.Magnitude
			if dist <= SW_STOP then
				return true
			end
			local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
			if not swAlive(id) or breakBlocked() then
				return false
			end
			local step = math.min(SW.Speed * dt, dist - SW_STOP)
			local np = r.Position + dir.Unit * step
			np = Vector3.new(np.X, 3.693, np.Z)
			r.CFrame = CFrame.new(np, Vector3.new(part.Position.X, np.Y, part.Position.Z))
			r.AssemblyLinearVelocity = Vector3.zero
		end
		return false
	end

	local function breakLoop(id)
		local lastEquip = 0
		reequipPickaxe()
		enableGameSwing()
		task.spawn(monitorSwing, id)
		while swAlive(id) do
			if breakBlocked() then
				SW.Working = false
				if not SW.Yielded then
					SW.Yielded = true
					setSwStatus(evBlock() and "Paused - Hell Event has priority" or (TT.Working and "Paused - Auto Titanic Egg is working (AutoSwing kept ON)" or "Paused - Auto Steal is working (AutoSwing kept ON)"))
				end
				task.wait(0.15)
			else
				if SW.Yielded then
					SW.Resuming = true
					SW.Yielded = false
					setSwStatus("Auto Steal finished -> equipping Pickaxe")

					local equipped = false
					for _ = 1, 5 do
						if not swAlive(id) then break end
						equipPickaxe()
						task.wait(0.25)
						local char = LocalPlayer.Character
						if char then
							for _, tool in ipairs(char:GetChildren()) do
								if isPickaxe(tool) then
									equipped = true
									break
								end
							end
						end
						if equipped then break end
					end

					if equipped and not SW.ManualOff then
						hookSwingEvents()
						local btn = ensureSwingRemote and ensureSwingRemote() or findSwingBtn()
						if btn then

							SW.EvtState = nil
							SW.Assumed = nil
							if readSwingState(btn) ~= true then

								pressBtn(btn)
								task.wait(0.35)
							end
						end
						setSwingEventState(true)
						SW.Assumed = true
						setSwStatus("Resumed - Pickaxe equipped, AutoSwing enabled")
					elseif not equipped then
						setSwStatus("Cannot resume - Pickaxe not found/equipped")
					end
					SW.Resuming = false
				elseif os.clock() - lastEquip > 1 then
					lastEquip = os.clock()
					equipPickaxe()
				end
				local e = scanEggs()[1]
				if not e then
					SW.Working = false
					setSwStatus("No matching egg found, scanning...")
					task.wait(0.5)
				else
					SW.Working = true
					TMX.Exit()
					Stats.SnapEgg(e)
					setSwStatus("Walking to egg (size " .. e.size .. ")")
					if swWalkTo(e.part, id) then
						setSwStatus("Breaking egg")
						while swAlive(id) and not breakBlocked() and eggAlive(e.part) do
							local r = getHRP()
							if r and Vector3.new(e.part.Position.X - r.Position.X, 0, e.part.Position.Z - r.Position.Z).Magnitude > SW_STOP * 4 then
								break
							end
							task.wait(0.3)
						end
						if swAlive(id) and not eggAlive(e.part) and not Stats.Seen[e.part] then
							Stats.Seen[e.part] = true
							Stats.AddEgg(Stats.SnapEgg(e))
						end
					end
				end
				task.wait()
			end
		end
	end

	local function swStop(msg)
		local was = SW.Running
		SW.Running = false
		SW.Working = false
		SW.Job += 1
		SW.Yielded = false
		if was then
			setGameSwing(false)
		end
		setSwingEventState(false)
		local hrp = getHRP()
		if hrp then
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
		if setSwStatus then
			setSwStatus(msg or "Stopped")
		end
	end

	local function swStart()
		local any = false
		for _, v in pairs(SW.Rarity) do
			if v then
				any = true
				break
			end
		end
		if not any then
			Notify("Select a rarity first", false)
			return false
		end
		SW.Running = true
		SW.Yielded = false
		SW.MonitorOff = false
		SW.ManualOff = false
		SW.Job += 1
		local id = SW.Job
		task.spawn(function()
			local ok, err = pcall(breakLoop, id)
			if not ok then
				warn("[NEXOEA Hub] AutoBreak error: " .. tostring(err))
				if SW.Job == id then
					swStop("Error - see console")
					pcall(function()
						swToggle.Set(false, true)
					end)
				end
			end
		end)
		return true
	end

	local setTtStatus
	local ttToggle, ttSpeedCtl, ttZoneDrop

	local function ttAlive(id)
		return TT.Running and TT.Job == id
	end

	local function titanicBlocked()
		return ST.Working or evBlock()
	end

	local function fmtClock(sec)
		sec = math.max(0, math.floor(sec))
		return ("%d:%02d"):format(math.floor(sec / 60), sec % 60)
	end

	local function isTitanicEgg(model, part)
		if part:GetAttribute("Titanic") == true then
			return true
		end
		local zi = tonumber(Workspace:GetAttribute("TitanicZoneIndex"))
		local nm = Workspace:GetAttribute("TitanicEggName")
		local ends = tonumber(Workspace:GetAttribute("TitanicEndsAt"))
		local despawn = tonumber(part:GetAttribute("DespawnAt"))
		if zi and nm and ends and despawn
			and tonumber(part:GetAttribute("ZoneIndex")) == zi
			and part:GetAttribute("EggType") == nm
			and math.abs(despawn - ends) < 2 then
			return true
		end
		return false
	end

	local function findTitanic()
		local build = Workspace:FindFirstChild("Build")
		local zb = build and build:FindFirstChild("ZoneBuilds")
		if not zb then
			return nil
		end
		local r = getHRP()
		local best, bd = nil, math.huge
		for _, zone in ipairs(zb:GetChildren()) do
			local eggs = zone:FindFirstChild("Eggs")
			local zn0 = tonumber(zone.Name:match("%d+"))
			if eggs then
				for _, m in ipairs(eggs:GetChildren()) do
					local part = m:FindFirstChild("Egg")
					local zn = (part and tonumber(part:GetAttribute("ZoneIndex"))) or zn0
					if part and part:IsA("BasePart") and (zn == nil or TT.Zone[zn]) and eggAlive(part)
						and isTitanicEgg(m, part) then
						local d = r and (part.Position - r.Position).Magnitude or 0
						if d < bd then
							best, bd = part, d
						end
					end
				end
			end
		end
		return best
	end

	local function ttStopDist(part)
		return math.max(SW_STOP, math.max(part.Size.X, part.Size.Z) / 2 + TT_GAP)
	end

	local function ttWalkTo(part, id)
		local stop = ttStopDist(part)
		if TT.TpMode then
			local done = TPX.reach({
				part = part,
				stop = stop,
				say = setTtStatus,
				alive = function()
					return ttAlive(id) and not titanicBlocked() and eggAlive(part)
				end,
			})
			if done then
				return true
			end
		end
		while ttAlive(id) and not titanicBlocked() and eggAlive(part) do
			local r = getHRP()
			if not r then
				return false
			end
			local dir = Vector3.new(part.Position.X - r.Position.X, 0, part.Position.Z - r.Position.Z)
			local dist = dir.Magnitude
			if dist <= stop then
				return true
			end
			local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
			if not ttAlive(id) or titanicBlocked() then
				return false
			end
			local step = math.min(TT.Speed * dt, dist - stop)
			local np = r.Position + dir.Unit * step
			np = Vector3.new(np.X, 3.693, np.Z)
			r.CFrame = CFrame.new(np, Vector3.new(part.Position.X, np.Y, part.Position.Z))
			r.AssemblyLinearVelocity = Vector3.zero
		end
		return false
	end

	local function ttPrepare()
		equipPickaxe()
		if EventGate.Active then
			return
		end
		local btn = findSwingBtn()
		local st = btn and readSwingState(btn)
		if st == true or (st == nil and SW.Assumed == true) then
			return
		end
		hookSwingEvents()
		TT.EnabledSwing = true
		enableGameSwing()
	end

	local function ttRelease()
		if TT.EnabledSwing and not SW.Running and not EventGate.Active then
			pcall(setGameSwing, false)
			pcall(setSwingEventState, false)
		end
		TT.EnabledSwing = false
	end

	local function ttHit(part, id)
		local r = getHRP()
		if not r then
			return
		end
		local holdCF = r.CFrame
		local lastEquip, lastChk, lastPress, lastStatus = 0, 0, 0, 0
		local wasHolding = nil
		while ttAlive(id) and not titanicBlocked() and eggAlive(part) do
			r = getHRP()
			if not r then
				break
			end
			r.CFrame = holdCF
			r.AssemblyLinearVelocity = Vector3.zero
			local now = os.clock()
			if now - lastEquip > 1 then
				lastEquip = now
				equipPickaxe()
			end
			if now - lastChk > 0.3 and not EventGate.Active then
				lastChk = now

				local holding = holdingPick()
				if wasHolding == false and holding then
					enableGameSwing()
				end
				wasHolding = holding
				if SW.EvtState == false then
					setSwingEventState(true)
				end
				local btn = findSwingBtn()
				if btn and readSwingState(btn) == false and now - lastPress > 0.8 then
					lastPress = now
					pressBtn(btn)
					SW.Assumed = true
				end
			elseif EventGate.Active then
				wasHolding = nil
			end
			if now - lastStatus > 0.5 then
				lastStatus = now
				local hp = tonumber(part:GetAttribute("Health")) or 0
				local mx = tonumber(part:GetAttribute("MaxHealth")) or hp
				setTtStatus(("Breaking Titanic Egg: %d / %d"):format(math.floor(hp), math.floor(mx)))
			end
			RunService.Heartbeat:Wait()
		end
	end

	local function titanicStep(id)
		do
			if next(TT.Zone) == nil then
				TT.Working = false
				setTtStatus("Select a zone first")
				task.wait(0.5)
			elseif titanicBlocked() then
				if not TT.Paused then
					TT.Paused = true
					TT.Working = false
					setTtStatus(evBlock() and "Paused - Hell Event has priority" or "Paused - Auto Steal is working")
				end
				task.wait(0.15)
			else
				TT.Paused = false
				local egg = findTitanic()
				if not egg then
					TT.Working = false
					local nextAt = Workspace:GetAttribute("TitanicNextAt")
					local cd = nextAt and (nextAt - os.time()) or 0
					setTtStatus("Waiting for Titanic Egg" .. (cd > 0 and (" (next in " .. fmtClock(cd) .. ")") or ""))
					task.wait(0.5)
				else
					if not TT.Working then
						TT.Working = true
						TMX.Exit()
						local t0 = os.clock()
						while ttAlive(id) and SW.Running and not SW.Yielded and os.clock() - t0 < 1.5 do
							RunService.Heartbeat:Wait()
						end
						if ttAlive(id) and not titanicBlocked() then
							setTtStatus("Preparing Pickaxe / AutoSwing")
							ttPrepare()
						end
					end
					if ttAlive(id) and not titanicBlocked() then
						Stats.SnapEgg({ part = egg, model = egg.Parent })
						setTtStatus("Walking to Titanic Egg")
						if ttWalkTo(egg, id) then
							ttHit(egg, id)
						end
						if ttAlive(id) and not titanicBlocked() and not eggAlive(egg) then
							if not Stats.Seen[egg] then
								Stats.Seen[egg] = true
								Stats.AddEgg(Stats.SnapEgg({ part = egg, model = egg.Parent }))
							end
							setTtStatus("Titanic Egg broken")
							Notify("Titanic Egg broken", true)
							TT.Working = false
							ttRelease()
							task.wait(1)
						end
					end
				end
			end
		end
	end

	local function titanicLoop(id)
		while ttAlive(id) do
			local ok, err = pcall(titanicStep, id)
			if not ok then
				warn("[NEXOEA Hub] AutoTitanicEgg: " .. tostring(err))
				task.wait(1)
			end
			task.wait()
		end
	end

	local function ttStart()
		TT.Running = true
		TT.Paused = false
		TT.Job += 1
		local id = TT.Job
		task.spawn(function()

				warn("[NEXOEA Hub] AutoTitanicEgg error (auto-restart): " .. tostring(err))
				TT.Working = false
				task.wait(1)
			end
		end)

	end

	local TM = { Running = false, Job = 0, Speed = 150, Margin = 7, GroundY = 3.693 }
	local setTmStatus, tmToggle

	function TM.Alive(id)
		return TM.Running and TM.Job == id
	end

	function TM.Find()
		local plots = Workspace:FindFirstChild("Plots")
		if not plots then
			return nil, "Plots not found"
		end
		local uid = LocalPlayer.UserId
		for _, plot in ipairs(plots:GetChildren()) do
			if tonumber(plot:GetAttribute("OwnerUserId")) == uid then
				if plot:GetAttribute("TreadmillUnlocked") == false then
					return nil, "Treadmill is locked"
				end
				for _, c in ipairs(plot:GetChildren()) do
					if c.Name ~= "TreadmillBoard" and c.Name:lower():find("treadmill", 1, true) then
						local hb = c:FindFirstChild("Hitbox")
						if hb and hb:IsA("BasePart") then
							return hb
						end
					end
				end
				return nil, "Treadmill not found in your plot"
			end
		end
		return nil, "Your plot was not found"
	end

	function TM.Inside(hb, hrp, margin)
		local m = margin or TM.Margin
		local rel = hb.CFrame:PointToObjectSpace(hrp.Position)
		return math.abs(rel.X) <= hb.Size.X / 2 + m
			and math.abs(rel.Z) <= hb.Size.Z / 2 + m
			and math.abs(rel.Y) <= hb.Size.Y / 2 + 6
	end

	function TM.Blocked()
		return ST.Working or TT.Working or SW.Working or evBlock()
	end

	TM.Leaving = false

	function TM.Leave(hb, hrp)
		if TM.Leaving then

			local w0 = os.clock()
			while TM.Leaving and os.clock() - w0 < 1.5 do
				task.wait(0.05)
			end
			return
		end
		TM.Leaving = true
		local ok, err = pcall(function()
			local hum = getHumanoid()
			if hum and hum.Health > 0 then
				local t0 = os.clock()
				while os.clock() - t0 < 0.3 do
					hum.Jump = true
					pcall(function() hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
					task.wait(0.05)
				end
			end
			local r = getHRP() or hrp
			if r and TM.Inside(hb, r) then
				local out = hb.CFrame:PointToWorldSpace(Vector3.new(hb.Size.X / 2 + TM.Margin + 4, 0, 0))
				r.CFrame = CFrame.new(out.X, TM.GroundY, out.Z)
				r.AssemblyLinearVelocity = Vector3.zero
			end
		end)
		TM.Leaving = false
		if not ok then
			warn("[NEXOEA Hub] Treadmill leave error: " .. tostring(err))
		end
	end

	TMX.Exit = function()
		if not TM.Running then return end
		local hb = TM.Find()
		local r = getHRP()
		if hb and r and TM.Inside(hb, r) then
			TM.Leave(hb, r)
		end
	end

	local function tmLoop(id)
		local touching = false
		while TM.Alive(id) do
			local hrp = getHRP()
			if not hrp then
				touching = false
				task.wait(0.3)
				continue
			end
			local hb, why = TM.Find()
			if not hb then
				touching = false
				setTmStatus(why)
				task.wait(1)
				continue
			end
			if TM.Blocked() then
				if touching or TM.Inside(hb, hrp) then
					TM.Leave(hb, hrp)
				end
				touching = false
				setTmStatus("Paused - another function is working")
				task.wait(0.1)
			elseif touching and TM.Inside(hb, hrp) then
				setTmStatus("On treadmill")
				task.wait(0.2)
			elseif TM.Inside(hb, hrp, 0) then
				touching = true
				setTmStatus("On treadmill")
				task.wait(0.2)
			else
				touching = false
				setTmStatus("Walking to treadmill")
				local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
				local r = getHRP()
				if TM.Alive(id) and r and not TM.Blocked() then
					local to = Vector3.new(hb.Position.X - r.Position.X, 0, hb.Position.Z - r.Position.Z)
					if to.Magnitude > 0.3 then
						local np = r.Position + to.Unit * math.min(TM.Speed * dt, to.Magnitude)
						np = Vector3.new(np.X, TM.GroundY, np.Z)
						r.CFrame = CFrame.new(np, np + Vector3.new(to.X, 0, to.Z))
						r.AssemblyLinearVelocity = Vector3.zero
					end
				end
			end
		end
	end

	function TM.Stop(msg)
		local was = TM.Running
		TM.Running = false
		TM.Job += 1
		if was then

			local hb = TM.Find()
			local hrp = getHRP()
			if hb and hrp and TM.Inside(hb, hrp) then
				TM.Leave(hb, hrp)
			end
		end
		if setTmStatus then
			setTmStatus(msg or "Stopped")
		end
	end

	function TM.Start()
		TM.Running = true
		TM.Job += 1
		local id = TM.Job
		task.spawn(function()
			while TM.Alive(id) do
				local ok, err = pcall(tmLoop, id)
				if ok then break end
				warn("[NEXOEA Hub] AutoTreadmill error (auto-restart): " .. tostring(err))
				task.wait(1)
			end
		end)
		return true
	end

		local function fmtNum(n)
			if type(n) ~= "number" then
				return "?"
			end
			for _, u in ipairs({ { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }) do
				if n >= u[1] then
					local str = string.format("%.2f", n / u[1]):gsub("0+$", ""):gsub("%.$", "")
					return str .. u[2]
				end
			end
			return tostring(math.floor(n))
		end

		local function afkStamp(sec)
			if type(sec) ~= "number" then
				return ""
			end
			sec = math.max(0, math.floor(sec))
			local th = tx("History") ~= "History"
			local uh, um, us = "h", "m", "s"
			if th then
				uh, um, us = "ชั่วโมง", "นาที", "วินาที"
			end
			local h = sec // 3600
			local m = (sec % 3600) // 60
			local sp = th and " " or ""
			if h > 0 then
				if m > 0 then
					return ("%d%s%s %d%s%s"):format(h, sp, uh, m, sp, um)
				end
				return ("%d%s%s"):format(h, sp, uh)
			end
			if m > 0 then
				return ("%d%s%s"):format(m, sp, um)
			end
			return ("%d%s%s"):format(sec, sp, us)
		end

		local function placeCam(v)
			local dir = Vector3.new(math.sin(v.angle) * 0.8, 0.45, math.cos(v.angle) * 0.8).Unit
			v.cam.CFrame = CFrame.lookAt(v.center + dir * v.dist, v.center)

		local function mountView(vf, src)
			if not src then
				return nil
			end
			local ok, clone = pcall(function()
				return src:Clone()
			end)
			if not ok or not clone then
				return nil
			end
			local holder = Instance.new("Model")
			clone.Parent = holder
			holder.Parent = vf
			local cf, size
			local okb, cf0, size0 = pcall(function()
				return holder:GetBoundingBox()
			end)
			if okb and cf0 and size0 and size0.Magnitude > 0 then
				cf, size = cf0, size0
			else
				-- fallback: measure the parts by hand
				local mn, mx
				for _, d in ipairs(holder:GetDescendants()) do
					if d:IsA("BasePart") then
						local h = d.Size / 2
						local lo, hi = d.Position - h, d.Position + h
						if mn then
							mn = Vector3.new(math.min(mn.X, lo.X), math.min(mn.Y, lo.Y), math.min(mn.Z, lo.Z))
							mx = Vector3.new(math.max(mx.X, hi.X), math.max(mx.Y, hi.Y), math.max(mx.Z, hi.Z))
						else
							mn, mx = lo, hi
						end
					end
				end
				if mn then
					cf, size = CFrame.new((mn + mx) / 2), mx - mn
				end
			end
			if not cf or not size or size.Magnitude <= 0 then
				holder:Destroy()
				return nil
			end
			local cam = Instance.new("Camera")
			cam.FieldOfView = 40
			cam.Parent = vf
			vf.CurrentCamera = cam
			local v = {
				cam = cam,
				center = cf.Position,
				dist = (size.Magnitude / 2) / math.tan(math.rad(20)) * 1.05,
				angle = math.random() * 6.28,
			}
			placeCam(v)
			-- meshes / textures that are not loaded yet are what made some icons show up empty
			task.spawn(function()
				pcall(function()
					game:GetService("ContentProvider"):PreloadAsync({ holder })
				end)
			end)
			return v
		end

			if ch == "" then
				ch = "?"
			end
			return ch:upper()
		end

	local rarityCache = setmetatable({}, { __mode = "k" })
	local function getRarity(obj)
		if rarityCache[obj] then
			return rarityCache[obj]
		end
		local res
		local cur = obj
		for _ = 1, 4 do
			if not cur then
				break
			end
			local v = cur:GetAttribute("Rarity")
			if typeof(v) == "string" and v ~= "" then
				res = v
				break
			end
			cur = cur.Parent
		end
		if not res then
			local nm = obj:GetAttribute("AnimalName") or obj.Name
			res = TemplateRarity[tostring(nm):lower()]
		end
		if not res then
			local rf = ReplicatedStorage:FindFirstChild("Rarities")
			if rf and rf:IsA("ModuleScript") then
				local ok, data = pcall(require, rf)
				if ok and type(data) == "table" then
					local e = data[obj.Name]
					if type(e) == "string" then
						res = e
					elseif type(e) == "table" then
						res = e.Rarity or e.rarity
					end
				end
			end
		end
		if not res then
			for _, d in ipairs(obj:GetDescendants()) do
				if (d:IsA("TextLabel") or d:IsA("TextButton")) and RANK[d.Text] then
					res = d.Text
					break
				end
				if d:IsA("UIGradient") and RANK[d.Name] then
					res = d.Name
					break
				end
				local v = d:GetAttribute("Rarity")
				if typeof(v) == "string" and v ~= "" then
					res = v
					break
				end
			end
		end
		if not res then
			local low = obj.Name:lower()
			for _, r in ipairs(STEAL_RARITIES) do
				if low:find(r:lower(), 1, true) then
					res = r
					break
				end
			end
		end
		if res then
			rarityCache[obj] = res
		end
		return res
	end

	local function zoneOfX(x)
		for z, rng in pairs(ZONE_X) do
			if x >= rng[1] - 5 and x <= rng[2] + 5 then
				return z
			end
		end
		local best, bd = nil, math.huge
		for z, rng in pairs(ZONE_X) do
			local d = math.min(math.abs(x - rng[1]), math.abs(x - rng[2]))
			if d < bd then
				best, bd = z, d
			end
		end
		return best
	end

	local function zoneIdx(pos)
		if pos.X > ZONE_X[1][2] + 5 then
			return 0
		end
		return zoneOfX(pos.X) or 0
	end

	local function zoneGap(a, b)
		return math.abs(zoneIdx(a) - zoneIdx(b))
	end

	local MONEY_MULT = { K = 1e3, M = 1e6, B = 1e9, T = 1e12 }
	local function toNumber(num, suffix)
		local n = tonumber((tostring(num):gsub(",", "")))
		if not n then return nil end
		return n * (MONEY_MULT[tostring(suffix or ""):upper()] or 1)
	end

	local function parseAmount(text)
		if type(text) ~= "string" then return nil end
		local num, suf = text:match("([%d][%d,%.]*)%s*([KkMmBbTt]?)")
		if not num then return nil end
		return toNumber(num, suf)
	end

	local function formatAmount(n)
		if type(n) ~= "number" or n <= 0 then return "" end
		for _, u in ipairs({ { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }) do
			if n >= u[1] then
				local str = string.format("%.2f", n / u[1]):gsub("0+$", ""):gsub("%.$", "")
				return str .. u[2]
			end
		end
		return tostring(math.floor(n))
	end

	local function incomeFromText(text)
		if type(text) ~= "string" or not text:find("/", 1, true) then return nil end
		text = text:gsub("<[^>]->", "")
		local num, suf = text:match("([%d][%d,%.]*)%s*([KkMmBbTt]?)%s*/%s*[sS]")
		if not num then return nil end
		return toNumber(num, suf)
	end
	local incomeLabel = setmetatable({}, { __mode = "k" })

	local function incomeOf(obj)
		local lbl = incomeLabel[obj]
		if lbl and lbl.Parent then
			local v = incomeFromText(lbl.Text)
			if v then return v end
		end
		for _, d in ipairs(obj:GetDescendants()) do
			if d:IsA("TextLabel") or d:IsA("TextButton") then
				local v = incomeFromText(d.Text)
				if v then
					incomeLabel[obj] = d
					return v
				end
			end
		end
		return nil
	end
	local function passIncome(obj)
		if ST.MinIncome <= 0 then return true end
		local inc = incomeOf(obj)
		if inc == nil then
			ST.Unread += 1
			return false
		end
		return inc >= ST.MinIncome
	end

	local MUT_NAMES = { "Gold", "Diamond", "Lucky" }
	local MUT_CANON = {}
	for _, n in ipairs(MUT_NAMES) do MUT_CANON[n:lower()] = n end

	local function mutationsOf(obj)
		local found, source = {}, nil
		local function addValue(v, from)
			if typeof(v) ~= "string" or v == "" then return end
			local canon = MUT_CANON[v:lower()]
			if canon then
				found[canon] = true
			else

				local hit = false
				for low, n in pairs(MUT_CANON) do
					if v:lower():find(low, 1, true) then found[n] = true; hit = true end
				end
				if not hit then found[v] = true end
			end
			source = source or from
		end
		local function scanAttrs(inst)
			if not inst then return end
			addValue(inst:GetAttribute("Variant"), "attr Variant")
			for k, v in pairs(inst:GetAttributes()) do
				local kl = tostring(k):lower()
				if k ~= "Variant" and (kl:find("mutation", 1, true) or kl:find("variant", 1, true)) then
					addValue(v, "attr " .. tostring(k))
				end
			end
		end
		scanAttrs(obj)
		scanAttrs(getPart(obj))
		if next(found) == nil then

			for _, c in ipairs(obj:GetChildren()) do
				if c:IsA("Model") or c:IsA("BasePart") then scanAttrs(c) end
			end
		end
		if next(found) == nil then

			for _, d in ipairs(obj:GetDescendants()) do
				if d.Name == "Mutation" and d:IsA("GuiObject") and d.Visible then
					for _, c in ipairs(d:GetChildren()) do
						if c:IsA("GuiObject") and c.Visible and MUT_CANON[c.Name:lower()] then
							found[MUT_CANON[c.Name:lower()]] = true
							source = source or "UI"
						end
					end
				end
			end
		end
		return found, source
	end

	local function chosenFrom(found)
		if next(ST.Mutation) == nil then
			return false
		end
		for n in pairs(ST.Mutation) do
			if found[n] then
				return true
			end
		end
		return false
	end
	local function hasChosenMutation(obj)
		return chosenFrom((mutationsOf(obj)))
	end

	local dbgSeen = {}

	local function getTargets(skip)
		local list, skipped = {}, 0
		ST.Unread = 0
		ST.Why = ""
		local folder = Workspace:FindFirstChild("AnimalPickups")
		if not folder then
			return list, skipped
		end
		local hrp = getHRP()
		local cut = { rarity = 0, norarity = 0, income = 0 }
		local total = 0
		for _, o in ipairs(folder:GetChildren()) do
			local part = getPart(o)
			if part then
				total += 1
				local r = getRarity(o)
				local z = zoneOfX(part.Position.X)
				local muts, msrc = mutationsOf(o)
				local hasMut = next(muts) ~= nil
				local chosen = chosenFrom(muts)
				local mutText = ""
				if hasMut then
					local names = {}
					for n in pairs(muts) do names[#names + 1] = n end
					table.sort(names)
					mutText = table.concat(names, "+") .. " (" .. tostring(msrc) .. ")"
				end

				local reason
				if not r then
					reason = "cut: rarity unreadable"
					cut.norarity += 1
				elseif not ST.Rarity[r] then
					reason = "cut: rarity " .. r .. " not selected"
					cut.rarity += 1
				elseif not (chosen or passIncome(o)) then
					reason = hasMut and ("cut: " .. mutText .. " not chosen and income below Min Income")
						or "cut: income below Min Income / unreadable"
					cut.income += 1
				end
				if hasMut then
					dbgLog(o, ("Variant=%s rarity=%s chosen=%s -> %s"):format(mutText, tostring(r), tostring(chosen), reason or "OK"))
				end
				if not reason then
					if skip[o] and skip[o] > os.clock() then
						skipped += 1
					else
						table.insert(list, {
							obj = o,
							part = part,
							rarity = r,
							rank = RANK[r] or 0,
							mut = chosen and 1 or 0,
							zone = z or 0,
							d = hrp and (part.Position - hrp.Position).Magnitude or 0,
						})
					end
				end
			end
		end
		if total > 0 and #list == 0 and skipped == 0 then
			ST.Why = ("%d animals: rarity not selected %d, income/mutation %d, rarity unreadable %d"):format(total, cut.rarity, cut.income, cut.norarity)
		end

		table.sort(list, function(a, b)
			if a.mut ~= b.mut then
				return a.mut > b.mut
			end
			if a.rank ~= b.rank then
				return a.rank > b.rank
			end
			return a.d < b.d
		end)
		return list, skipped
	end

	local function sleep(t, id)
		local t0 = os.clock()
		while os.clock() - t0 < t do
			if not stAlive(id) then
				return false
			end
			RunService.Heartbeat:Wait()
		end
		return stAlive(id)
	end

	local function moveTo(target, id, stopDist, abortFn, forceWalk, abortEvery, speed)
		stopDist = stopDist or 2.5
		local spd = speed or ST.Speed
		local nextCheck = 0
		local function resolve()
			if type(target) == "function" then
				return target()
			end
			return target
		end
		local hrp = getHRP()
		local pos = resolve()
		if not hrp or not pos then
			return false
		end
		if (ST.TpMode or ST.Ext) and ST.TpArmed and not forceWalk then

			if not stAlive(id) or EventGate.Abort then
				return false
			end
			local np = Vector3.new(pos.X, GROUND_Y, pos.Z)
			hrp.CFrame = CFrame.new(np)
			hrp.AssemblyLinearVelocity = Vector3.zero
			RunService.Heartbeat:Wait()
			return true
		end
		local t0 = os.clock()
		local off0 = Vector3.new(pos.X - hrp.Position.X, 0, pos.Z - hrp.Position.Z).Magnitude
		local timeout = off0 / math.max(spd, 1) * 3 + 8
		while stAlive(id) and not EventGate.Abort do
			hrp = getHRP()
			pos = resolve()
			if not hrp or not pos then
				return false
			end
			local off = Vector3.new(pos.X - hrp.Position.X, 0, pos.Z - hrp.Position.Z)
			if off.Magnitude <= stopDist then
				return true
			end
			if os.clock() - t0 > timeout then
				return false
			end
			if abortFn and os.clock() > nextCheck then
				nextCheck = os.clock() + (abortEvery or 0.4)
				if abortFn() then
					return false
				end
			end
			local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
			if not stAlive(id) then
				return false
			end
			local step = math.min(spd * dt, off.Magnitude)
			local np = hrp.Position + off.Unit * step
			np = Vector3.new(np.X, GROUND_Y, np.Z)
			hrp.CFrame = CFrame.new(np, np + Vector3.new(off.X, 0, off.Z))
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
		return false
	end

	local function isStolen(obj, before)
		if not obj.Parent then
			return true
		end
		local f = Workspace:FindFirstChild("CarriedAnimals")
		if not f then
			return false
		end
		if obj:IsDescendantOf(f) then
			return true
		end
		if #f:GetDescendants() > before then
			for _, d in ipairs(f:GetDescendants()) do
				if d.Name == obj.Name then
					return true
				end
			end
		end
		return false
	end

	local function findPrompt(obj, part, allowGlobal)
		local found
		for _, d in ipairs(obj:GetDescendants()) do
			if d:IsA("ProximityPrompt") then
				if d.ActionText == "Steal" then
					return d
				end
				found = found or d
			end
		end
		if obj:IsA("ProximityPrompt") then
			return obj
		end
		if found then
			return found
		end
		if allowGlobal and part then
			local best, bd = nil, 15
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("ProximityPrompt") and (d.ActionText == "Steal" or d.ObjectText == obj.Name) then
					local p = d.Parent
					local pos = p and (p:IsA("Attachment") and p.WorldPosition or p:IsA("BasePart") and p.Position)
					if pos then
						local dist = (pos - part.Position).Magnitude
						if dist < bd then
							best, bd = d, dist
						end
					end
				end
			end
			return best
		end
	end

	local function promptPos(prompt, part)
		local p = prompt and prompt.Parent
		if p then
			if p:IsA("Attachment") then
				return p.WorldPosition
			end
			if p:IsA("BasePart") then
				return p.Position
			end
		end
		return part and part.Position
	end

	local function firePrompt(prompt, attempt)
		pcall(function()
			prompt.Enabled = true
			prompt.MaxActivationDistance = 60
			prompt.RequiresLineOfSight = false
			prompt.HoldDuration = 0
		end)
		if fireproximityprompt and attempt % 3 ~= 0 then
			pcall(fireproximityprompt, prompt)
		else
			pcall(function()
				prompt:InputHoldBegin()
				task.wait(0.2)
				prompt:InputHoldEnd()
			end)
		end
	end

	local function groundClamp(v)
		return Vector3.new(v.X, math.clamp(v.Y, GROUND_Y, GROUND_Y + 5), v.Z)
	end

	local function verifyTarget(o)
		if not (o and o.Parent) then
			return false
		end
		local pf = Workspace:FindFirstChild("AnimalPickups")
		if pf and not o:IsDescendantOf(pf) then
			return false
		end
		local r = getRarity(o)
		if not r or not ST.Rarity[r] then
			return false
		end
		if chosenFrom((mutationsOf(o))) then
			return true
		end
		return passIncome(o)
	end

	local function findMine(obj, beforeSet)
		local f = Workspace:FindFirstChild("CarriedAnimals")
		if not f then
			return nil
		end
		if obj.Parent and obj:IsDescendantOf(f) then
			return obj
		end
		local fallback
		for _, d in ipairs(f:GetDescendants()) do
			if not beforeSet[d] then
				if d.Name == obj.Name then
					return d
				end
				fallback = fallback or d
			end
		end
		return fallback
	end

	local function collect(obj, part, id, skipVerify)
		cleanupLock()
		local lockPos
		lockConn = RunService.Heartbeat:Connect(function()
			if not (ST.Running or ST.Ext) then
				cleanupLock()
				return
			end
			local hrp = getHRP()
			if hrp and lockPos then
				hrp.CFrame = CFrame.new(lockPos)
				hrp.AssemblyLinearVelocity = Vector3.zero
			end
		end)

		local before = carriedCount()
		local beforeSet = snapshotCarried()
		local result = false
		local invalid = false
		local t0, tries, lastGlobal = os.clock(), 0, 0
		while stAlive(id) and os.clock() - t0 < 10 do
			if isStolen(obj, before) then
				result = true
				break
			end
			if not skipVerify and not verifyTarget(obj) then
				invalid = true
				break
			end
			local allowGlobal = os.clock() - lastGlobal > 1.5
			if allowGlobal then
				lastGlobal = os.clock()
			end
			local prompt = findPrompt(obj, part, allowGlobal)
			if prompt then
				local pp = promptPos(prompt, part)
				lockPos = pp and groundClamp(pp) or lockPos
				tries += 1
				stStatus(("Stealing %s (try %d)"):format(obj.Name, tries))
				if tries == 1 then
					sleep(0.25, id)
				end
				if not stAlive(id) then
					break
				end
				firePrompt(prompt, tries)
			else
				lockPos = (part and part.Parent and groundClamp(part.Position)) or lockPos
				stStatus("Steal prompt not found for " .. obj.Name)
			end
			if not sleep(0.4, id) then
				break
			end
		end
		if not result and stAlive(id) then
			result = isStolen(obj, before)
		end
		cleanupLock()
		local mine
		if result then
			mine = findMine(obj, beforeSet)
		end
		return result, mine, invalid
	end

	local function claimWork(id)
		ST.Working = true
		TMX.Exit()
		local t0 = os.clock()
		while stAlive(id) and ((SW.Running and not SW.Yielded) or TT.Working) and os.clock() - t0 < 1.5 do
			RunService.Heartbeat:Wait()
		end
	end

	local guardCache, guardAt = nil, 0
	local function nearestGuard()
		local hrp = getHRP()
		if not hrp then
			return nil
		end
		if not (guardCache and os.clock() - guardAt < 10) then
			guardCache, guardAt = {}, os.clock()
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d.Name == "GuardOverhead" then
					local p = d
					while p and not p:IsA("BasePart") do
						p = p.Parent
					end
					if p then
						table.insert(guardCache, p)
					end
				end
			end
		end
		local best, bd
		for _, p in ipairs(guardCache) do
			if p.Parent then
				local dist = (p.Position - hrp.Position).Magnitude
				if not bd or dist < bd then
					best, bd = p, dist
				end
			end
		end
		return best
	end

	-- Teleport mode, outbound leg. From 2+ zones away from the target we break an egg of the zone we stand in
	-- (Safe Zone -> zone 1), pick the hatched animal up and let that zone's guard hit us, which arms the teleport.
	ST.PrimeSpeed = 50

	function ST.PrimeZone(pos)
		local z = zoneIdx(pos)
		if z >= 1 then
			return z
		end
		local d1 = Vector3.new(pos.X - SAFE.X, 0, pos.Z - SAFE.Z).Magnitude
		local d2 = Vector3.new(pos.X - START.X, 0, pos.Z - START.Z).Magnitude
		if math.min(d1, d2) <= 80 then
			return 1
		end
		return nil
	end

	local function primeFolder(n)
		local build = Workspace:FindFirstChild("Build")
		local zb = build and build:FindFirstChild("ZoneBuilds")
		return zb and zb:FindFirstChild("Zone" .. n)
	end

	function ST.FindPrimeEgg(zone)
		local hrp = getHRP()
		if not hrp then
			return nil
		end
		local function nearestIn(n)
			local zf = primeFolder(n)
			local eggs = zf and zf:FindFirstChild("Eggs")
			if not eggs then
				return nil
			end
			-- smallest egg wins (same eggSize the stats use); equal size -> the nearer one
			local best, bs, bd
			for _, m in ipairs(eggs:GetChildren()) do
				local part = m:FindFirstChild("Egg")
				if part and part:IsA("BasePart") and eggAlive(part) then
					local sz = eggSize(m, part)
					local d = (part.Position - hrp.Position).Magnitude
					if not bs or sz < bs or (sz == bs and d < bd) then
						best, bs, bd = { part = part, model = m, zone = n }, sz, d
					end
				end
			end
			return best, bd
		end
		return (nearestIn(zone))
	end

	-- the guard that belongs to a zone (they live inside the same ZoneN folder as the eggs)
	local zoneGuardCache = {}
	function ST.ZoneGuard(n)
		local cached = zoneGuardCache[n]
		if cached and cached.Parent then
			return cached
		end
		local zf = primeFolder(n)
		if not zf then
			return nil
		end
		for _, d in ipairs(zf:GetDescendants()) do
			if d.Name == "GuardOverhead" then
				local pt = d
				while pt and not pt:IsA("BasePart") do
					pt = pt.Parent
				end
				if not pt then
					local mdl = d:FindFirstAncestorWhichIsA("Model")
					pt = mdl and getPart(mdl)
				end
				if pt then
					zoneGuardCache[n] = pt
					return pt
				end
			end
		end
		return nil
	end

	function ST.EggPrime(id, zone)
		local egg = ST.FindPrimeEgg(zone)
		if not egg then
			stStatus("Teleport: no egg in this zone -> normal route")
			return false
		end
		local part, model = egg.part, egg.model
		local epos = part.Position
		local swingOwned = not SW.Running
		local function cleanup()
			if swingOwned then
				pcall(setGameSwing, false)
				pcall(setSwingEventState, false)
			end
		end
		local function toEgg()
			return moveTo(function()
				return part.Parent and part.Position or nil
			end, id, SW_STOP, function()
				return not eggAlive(part)
			end, true, 0.2, ST.PrimeSpeed)
		end

		local pickBefore = {}
		local pf0 = Workspace:FindFirstChild("AnimalPickups")
		if pf0 then
			for _, c in ipairs(pf0:GetChildren()) do
				pickBefore[c] = true
			end
		end
		local eggInfo = {
			part = part,
			model = model,
			name = model.Name,
			size = eggSize(model, part),
			rarityName = eggRarity(model),
		}
		Stats.SnapEgg(eggInfo)

		stStatus("Teleport: walking to an egg")
		equipPickaxe()
		toEgg()
		if not stAlive(id) then
			cleanup()
			return false
		end

		stStatus("Teleport: breaking the egg")
		equipPickaxe()
		enableGameSwing()
		local t0, lastKeep = os.clock(), os.clock()
		while stAlive(id) and eggAlive(part) and os.clock() - t0 < 40 do
			if os.clock() - lastKeep > 1 then
				lastKeep = os.clock()
				equipPickaxe()
				local btn = findSwingBtn()
				if btn and readSwingState(btn) == false then
					pressBtn(btn)
				end
				local r = getHRP()
				if r and Vector3.new(part.Position.X - r.Position.X, 0, part.Position.Z - r.Position.Z).Magnitude > SW_STOP * 2 then
					toEgg()
				end
			end
			RunService.Heartbeat:Wait()
		end
		if not stAlive(id) then
			cleanup()
			return false
		end
		if eggAlive(part) then
			cleanup()
			stStatus("Teleport: egg not broken -> normal route")
			return false
		end
		if not Stats.Seen[part] then
			Stats.Seen[part] = true
			Stats.AddEgg(Stats.SnapEgg(eggInfo))
		end

		stStatus("Teleport: picking up the animal")
		local obj, apart
		local t1 = os.clock()
		repeat
			local pf = Workspace:FindFirstChild("AnimalPickups")
			if pf then
				local bd
				for _, c in ipairs(pf:GetChildren()) do
					if not pickBefore[c] then
						local pt = getPart(c)
						if pt then
							local d = Vector3.new(pt.Position.X - epos.X, 0, pt.Position.Z - epos.Z).Magnitude
							if d <= 80 and (not bd or d < bd) then
								obj, apart, bd = c, pt, d
							end
						end
					end
				end
			end
			if not obj then
				RunService.Heartbeat:Wait()
			end
		until obj or os.clock() - t1 > 8 or not stAlive(id)
		if not (obj and apart) then
			cleanup()
			if stAlive(id) then
				stStatus("Teleport: no animal from the egg -> normal route")
			end
			return false
		end
		moveTo(function()
			return apart.Parent and apart.Position or nil
		end, id, 3, nil, true, 0.2, ST.PrimeSpeed)
		local okc, mine = false, nil
		if stAlive(id) and apart.Parent then
			okc, mine = collect(obj, apart, id, true)
		end
		cleanup()
		if not stAlive(id) or not okc then
			return false
		end

		local function holding()
			local cf = Workspace:FindFirstChild("CarriedAnimals")
			if not cf then
				return false
			end
			if mine then
				return mine.Parent ~= nil and mine:IsDescendantOf(cf)
			end
			return carriedCount() > 0
		end
		for _ = 1, 3 do
			if not stAlive(id) or not holding() then
				break
			end
			local g = ST.ZoneGuard(egg.zone) or nearestGuard()
			if g then
				stStatus("Teleport: walking to the guard")
				moveTo(function()
					return g.Parent and g.Position or nil
				end, id, 8, function()
					return not holding()
				end, true, 0.05, ST.PrimeSpeed)
			end
			local t2 = os.clock()
			while stAlive(id) and os.clock() - t2 < 3 and holding() do
				RunService.Heartbeat:Wait()
			end
		end
		if not stAlive(id) then
			return false
		end
		if holding() then
			return false
		end
		stStatus("Teleport: guard hit -> teleporting to the target")
		return true
	end

	local function flatDist(a, b)
		return Vector3.new(a.X - b.X, 0, a.Z - b.Z).Magnitude
	end

	local function tpReach(target, id, stopDist)
		local reachDist = stopDist + 9
		for attempt = 1, 3 do
			if not stAlive(id) then
				return false
			end
			local pos = target()
			local hrp = getHRP()
			if not pos or not hrp then
				return false
			end
			if flatDist(hrp.Position, pos) <= reachDist then
				return true
			end
			if attempt > 1 then
				stStatus("Teleport: retrying to reach the target")
			end
			ST.TpArmed = true
			moveTo(target, id, stopDist)
			RunService.Heartbeat:Wait()
			hrp = getHRP()
			pos = target()
			if hrp and pos and flatDist(hrp.Position, pos) <= reachDist then
				return true
			end
			sleep(0.15, id)
		end
		if not stAlive(id) then
			return false
		end
		stStatus("Teleport failed -> walking")
		ST.TpArmed = false
		return moveTo(target, id, stopDist, nil, true)
	end

	-- Teleport move mode for other features. Returns true when we ended up at the target.
	-- Same recipe as Auto Steal: from 2+ zones away break an egg here, get hit by the guard, then teleport.
	-- o = { part, stop, alive, say }
	function TPX.reach(o)
		if ST.Ext or (ST.Running and (ST.Busy or ST.Working)) then
			return false
		end
		local hrp = getHRP()
		local part = o.part
		if not (hrp and part and part.Parent and o.alive()) then
			return false
		end
		local zone = ST.PrimeZone(hrp.Position)
		if not (zone and zoneGap(hrp.Position, part.Position) >= 2) then
			return false
		end
		local id = {}
		ST.Ext = { id = id, alive = o.alive, say = o.say or function() end }
		local reached = false
		pcall(function()
			local primed = ST.EggPrime(id, zone)
			if not (primed and o.alive() and part.Parent) then
				return
			end
			local stop = math.max(o.stop or 4, 2)
			local from = getHRP() and getHRP().Position or part.Position
			local dir = Vector3.new(from.X - part.Position.X, 0, from.Z - part.Position.Z)
			dir = dir.Magnitude > 0.1 and dir.Unit or Vector3.new(0, 0, 1)
			local function target()
				if not part.Parent then
					return nil
				end
				return part.Position + dir * math.max(stop - 1, 1)
			end
			ST.TpArmed = true
			reached = tpReach(target, id, stop)
		end)
		ST.TpArmed = false
		ST.Ext = nil
		return reached
	end

	local function returnToSpot(id, pos, armed)
		if not stAlive(id) or not pos then
			return
		end
		local hrp = getHRP()
		if not hrp then
			return
		end
		local function target()
			return pos
		end
		if armed and zoneGap(hrp.Position, pos) >= 2 then
			stStatus("Teleport -> back to the original spot")
			tpReach(target, id, 2.5)
		else
			stStatus("Walking back to the original spot")
			ST.TpArmed = false
			moveTo(target, id, 2.5, nil, true)
		end
		ST.TpArmed = false
	end

	local function runAuto(id)
		local skip = {}
		local needSafe = true
		local startFromSafe = true
		local tpHitDone = false
		local stealPos = nil
		local function hasTarget()
			local l = getTargets(skip)
			return #l > 0
		end

		while stAlive(id) do
			if evBlock() then

				if ST.Working or ST.Busy then
					ST.Working = false
					setBusy(false)
				end
				startFromSafe = false
				needSafe = false
				stStatus(EventGate.Active and "Paused - Hell Event has priority" or "Paused - still in Hell arena")
				sleep(0.4, id)
				continue
			end
			local hrp = getHRP()
			if not hrp then
				sleep(0.5, id)
				continue
			end

			if not Workspace:FindFirstChild("AnimalPickups") then
				stStatus("Waiting for AnimalPickups folder...")
				sleep(1, id)
				continue
			end

			local list, skipped = getTargets(skip)
			local e = list[1]

			if not e then

				ST.Working = false
				startFromSafe = true
				setBusy(false)
				if skipped > 0 then
					stStatus("Waiting for next round (some skipped)")
					sleep(0.6, id)
				elseif ST.SafeWait and needSafe then
					local flat = Vector3.new(hrp.Position.X - SAFE.X, 0, hrp.Position.Z - SAFE.Z).Magnitude
					if flat > 4 and not inAnyZone(hrp.Position) then
						claimWork(id)
						stStatus("No target -> going to Safe Zone")
						setBusy(true)
						moveTo(SAFE, id, 2.5, hasTarget)
						setBusy(false)
						ST.Working = false
					end
					needSafe = false
				else
					stStatus(ST.Unread > 0 and ("Cannot read income of " .. ST.Unread .. " animal(s) yet") or (ST.Why ~= "" and ST.Why or "Waiting for animals"))
					sleep(0.6, id)
				end
				continue
			end

			if not ST.Working then
				claimWork(id)
				if not stAlive(id) then
					break
				end
			end
			ST.TpArmed = false
			stealPos = nil
			local animalZone = zoneOfX(e.part.Position.X)
			needSafe = true
			setBusy(true)

			if startFromSafe then
				startFromSafe = false
				local flat = Vector3.new(hrp.Position.X - SAFE.X, 0, hrp.Position.Z - SAFE.Z).Magnitude
				if flat > 6 and not inAnyZone(hrp.Position) then
					stStatus("Going to Safe Zone first")
					moveTo(SAFE, id, 2.5)
					continue
				end
			end

			if ST.TpMode and not tpHitDone then
				local hp0 = getHRP()
				local pz = hp0 and ST.PrimeZone(hp0.Position)
				if pz and zoneGap(hp0.Position, e.part.Position) >= 2 then
					local primed = ST.EggPrime(id, pz)
					if not stAlive(id) then
						break
					end
					ST.TpArmed = primed == true
					if not e.part.Parent then
						ST.TpArmed = false
						continue
					end
				end
			end

			stStatus(("Going to %s [%s] zone %d"):format(e.obj.Name, e.rarity, e.zone))
			local function toAnimal()
				return e.part.Parent and e.part.Position or nil
			end
			if ST.TpMode and ST.TpArmed then
				tpReach(toAnimal, id, 3)
				ST.TpArmed = false
			else
				moveTo(toAnimal, id, 3)
			end
			if not stAlive(id) then
				break
			end

			local eMuts = mutationsOf(e.obj)
			local eInfo = nil
			if Stats.On then
				local obj = e.obj
				local nm = obj:GetAttribute("AnimalName") or obj:GetAttribute("DisplayName") or obj.Name
				eInfo = {
					Name = tostring(nm),
					Rarity = e.rarity,
					Income = incomeOf(obj),
					Model = Stats.ViewClone(obj),
				}
			end
			local ok, mine, invalid = false, nil, false
			local pickBefore = {}
			do
				local pf = Workspace:FindFirstChild("AnimalPickups")
				if pf then
					for _, c in ipairs(pf:GetChildren()) do
						pickBefore[c] = true
					end
				end
			end
			for attempt = 1, 3 do
				if not stAlive(id) or not e.part.Parent or EventGate.Abort then
					break
				end
				if attempt > 1 then
					stStatus(("Retry %d: %s"):format(attempt, e.obj.Name))
					moveTo(function()
						return e.part.Parent and e.part.Position or nil
					end, id, 3)
				end
				ok, mine, invalid = collect(e.obj, e.part, id)
				if ok or invalid then
					break
				end
			end
			if not stAlive(id) then
				break
			end

			if ok then
				ST.Collected += 1
				local hs = getHRP()
				stealPos = hs and hs.Position or nil
				local deliverTp = ST.TpMode and animalZone >= 2
				local histEntry = Stats.AddSteal(eMuts, eInfo or {})
				stStatus("Stolen! Heading back to START")
				local DROP_NEAR_START = 30
				local DROP_SEARCH_RADIUS = 80
				local GUARD_HIT_RADIUS = 20
				local curObj, curPart = e.obj, e.part
				local recoveries, guardTries, gone, finished = 0, 0, false, false

				local function holding()
					if not mine then
						return false
					end
					local cf = Workspace:FindFirstChild("CarriedAnimals")
					return cf ~= nil and mine.Parent ~= nil and mine:IsDescendantOf(cf)
				end
				local function lost()
					if not mine or holding() then
						return false
					end
					local h = getHRP()
					if h and Vector3.new(h.Position.X - START.X, 0, h.Position.Z - START.Z).Magnitude <= DROP_NEAR_START then
						return false
					end
					return true
				end
				local function byGuard()
					if flingRemote then
						local t = os.clock()
						while os.clock() - t < 0.3 and os.clock() - flingAt > 1.5 do
							RunService.Heartbeat:Wait()
						end
						return os.clock() - flingAt <= 1.5
					end
					local g = nearestGuard()
					local h = getHRP()
					return g ~= nil and h ~= nil and (g.Position - h.Position).Magnitude <= GUARD_HIT_RADIUS
				end
				local function findDropped()
					local pf = Workspace:FindFirstChild("AnimalPickups")
					if not pf then
						return nil
					end
					if curObj.Parent == pf then
						local p = getPart(curObj)
						if p then
							return curObj, p
						end
					end
					local h = getHRP()
					local best, bp, bd
					for _, c in ipairs(pf:GetChildren()) do
						if not pickBefore[c] and c.Name == curObj.Name then
							local p = getPart(c)
							if p then
								local d = h and (p.Position - h.Position).Magnitude or 0
								if d <= DROP_SEARCH_RADIUS and (not bd or d < bd) then
									best, bp, bd = c, p, d
								end
							end
						end
					end
					return best, bp
				end
				local function awaitDropped(sec)
					local t1 = os.clock()
					repeat
						local o, p = findDropped()
						if o and p then
							return o, p
						end
						RunService.Heartbeat:Wait()
					until os.clock() - t1 >= sec or not stAlive(id)
					return nil
				end
				local function recover(hitByGuard)
					local obj, part = awaitDropped(2.5)
					if not (obj and part) then
						return false
					end
					stStatus("Animal dropped -> picking it up again")
					local armed = ST.TpArmed
					ST.TpArmed = ST.TpMode and hitByGuard
					moveTo(function()
						return part.Parent and part.Position or nil
					end, id, 3)
					ST.TpArmed = armed
					if not stAlive(id) or not part.Parent then
						return false
					end
					local ok2, mine2 = collect(obj, part, id, true)
					if ok2 then
						mine, curObj, curPart = mine2, obj, part
						return true
					end
					return false
				end

				while stAlive(id) and not gone and not finished do
					if deliverTp and not tpHitDone and guardTries < 3 and mine then
						ST.TpArmed = false
						guardTries += 1
						if stAlive(id) and not lost() then
							stStatus("Stolen! (Teleport) going to the guard")
						end
						if stAlive(id) and not lost() then
							local g = ST.ZoneGuard(animalZone) or nearestGuard()
							if g then
								stStatus("Walking to the guard")
								moveTo(function()
									return g.Parent and g.Position or nil
								end, id, 8, lost, true, 0.05, ST.PrimeSpeed)
								local t2 = os.clock()
								while stAlive(id) and os.clock() - t2 < 3 and not lost() do
									RunService.Heartbeat:Wait()
								end
							end
						end
						if stAlive(id) and lost() then
							local gh = byGuard()
							recoveries += 1
							if recoveries > 5 or not recover(gh) then
								gone = true
							else
								tpHitDone = gh
							end
						elseif stAlive(id) then
							guardTries = 3
						end
					elseif deliverTp and tpHitDone then
						ST.TpArmed = true
						stStatus("Teleport -> Safe Zone")
						moveTo(SAFE, id, 2.5)
						local t3 = os.clock()
						local safeWait = 2.5
						while stAlive(id) and os.clock() - t3 < safeWait and holding() do
							RunService.Heartbeat:Wait()
						end
						if stAlive(id) and holding() then
							stStatus("Teleport -> START (delivering)")
							moveTo(START, id, 2.5)
							local t4 = os.clock()
							while stAlive(id) and os.clock() - t4 < 2.5 and holding() do
								RunService.Heartbeat:Wait()
							end
						end
						if not stAlive(id) or holding() then
							finished = true
						elseif awaitDropped(0.6) then
							local gh = byGuard()
							recoveries += 1
							if recoveries > 5 or not recover(gh) then
								gone = true
							else
								tpHitDone = gh
							end
						else
							finished = true
						end
					else
						ST.TpArmed = false
						local rp = roadPoint(project(curPart.Position))
						moveTo(Vector3.new(rp.X, GROUND_Y, rp.Z), id, nil, lost, false, 0.05)
						if stAlive(id) and not lost() then
							moveTo(START, id, nil, lost, false, 0.05)
						end
						if not stAlive(id) then
							finished = true
						elseif lost() then
							local gh = byGuard()
							recoveries += 1
							if recoveries > 5 or not recover(gh) then
								gone = true
							elseif ST.TpMode then
								tpHitDone = gh
							end
						else
							finished = true
						end
					end
				end

				if gone then
					ST.Collected = math.max(0, ST.Collected - 1)
					Stats.UndoSteal(histEntry)
					stStatus("Animal lost -> next target")
					if stAlive(id) and ST.TpMode and stealPos then
						returnToSpot(id, stealPos, tpHitDone)
					end
					stealPos = nil
					tpHitDone = false
					ST.TpArmed = false
					needSafe = false
					startFromSafe = false
				elseif ST.TpMode and stealPos then
					if stAlive(id) then
						returnToSpot(id, stealPos, tpHitDone)
					end
					stealPos = nil
					tpHitDone = false
					ST.TpArmed = false
					needSafe = false
					startFromSafe = false
				else
					stealPos = nil
					tpHitDone = false
					ST.TpArmed = false
					if stAlive(id) and ST.SafeWait then
						stStatus("At START -> going to Safe Zone")
						moveTo(SAFE, id)
						needSafe = false
					end
					startFromSafe = true
				end
			else
				skip[e.obj] = os.clock() + 15
				needSafe = false
				stStatus(invalid and "Target no longer matches - skipped" or "Steal failed -> next target")
			end
		end
		setBusy(false)
		ST.Working = false
	end

	local function stStop(msg)
		ST.Running = false
		ST.TpArmed = false
		ST.Working = false
		stJob += 1
		cleanupLock()
		setBusy(false)
		restoreCollide()
		restoreRagdollStates()
		local hrp = getHRP()
		if hrp then
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
		stStatus(msg or "Stopped")
	end

	local function stStart()
		ST.Running = true
		stJob += 1
		local id = stJob
		task.spawn(function()
			local ok, err = pcall(runAuto, id)
			if not ok then
				warn("[NEXOEA Hub] AutoSteal error: " .. tostring(err))
				if stJob == id then
					stStop("Error - see console")
					pcall(function()
						stToggle.Set(false, true)
					end)
				end
			end
		end)
		return true
	end

	local WATCH = { HellJoinRemote = true }
	local Learned = {}
	if hookmetamethod and getnamecallmethod then
		pcall(function()
			local old
			local hook = function(self, ...)
				if getnamecallmethod() == "FireServer" and typeof(self) == "Instance"
					and self.Parent == ReplicatedStorage and WATCH[self.Name] then
					Learned[self.Name] = table.pack(...)
				end
				return old(self, ...)
			end
			old = hookmetamethod(game, "__namecall", newcclosure and newcclosure(hook) or hook)
		end)
	end

	local function fire(name, target)
		local r = ReplicatedStorage:FindFirstChild(name)
		if not r then return false end
		local l = Learned[name]
		local args, n
		if l then
			args, n = table.clone(l), l.n
			if target then
				for i = 1, n do
					if typeof(args[i]) == "Instance" then args[i] = target break end
				end
			end
		else
			args = target and { target } or {}
			n = #args
		end
		pcall(function()
			if r:IsA("RemoteEvent") then
				r:FireServer(table.unpack(args, 1, n))
			elseif r:IsA("RemoteFunction") then
				task.spawn(r.InvokeServer, r, table.unpack(args, 1, n))
			end
		end)
		return true
	end

			end
		end
	end

	local function floorY(r)
		local char = r.Parent
		local ignore = { char }
		local a = arena()
		if a then
			local eg = a:FindFirstChild("EssenceEggs")
			if eg then table.insert(ignore, eg) end
			local b = a:FindFirstChild("DemonBoss")
			if b then table.insert(ignore, b) end
		end
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = ignore
		local hit
		for _ = 1, 3 do
			hit = Workspace:Raycast(r.Position + Vector3.new(0, 2, 0), Vector3.new(0, -600, 0), params)
			if hit and hit.Instance.Name == "Egg" then
				table.insert(ignore, hit.Instance)
				params.FilterDescendantsInstances = ignore
				hit = nil
			else
				break
			end
		end
		if not hit then return nil end
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local hip = hum and hum.HipHeight or 2
		return hit.Position.Y + hip + r.Size.Y / 2
	end

	local function settleY(r, step)
		local cur = r.Position.Y
		local hum = r.Parent and r.Parent:FindFirstChildOfClass("Humanoid")
		if hum and hum.FloorMaterial ~= Enum.Material.Air then return cur end
		local down = cur - (step or 3)
		local fy = floorY(r)
		if fy then return math.max(fy, down) end
		return down
	end

	local function walkTo(pos, gen, stopDist, cond)
		stopDist = stopDist or 5

		local r0 = getRoot()
		if not r0 then return false end
		local highArena = r0.Position.Y > 500
		local gy = highArena and r0.Position.Y or 3.693
		while alive(gen) and (not cond or cond()) do
			local r = getRoot()
			if not r then return false end
			local dir = Vector3.new(pos.X - r.Position.X, 0, pos.Z - r.Position.Z)
			local dist = dir.Magnitude
			if dist <= stopDist then return true end
			local dt = math.min(RunService.Heartbeat:Wait(), 0.05)
			if not alive(gen) then return false end
			local step = math.min(EV.Speed * dt, dist - stopDist)
			local np = r.Position + dir.Unit * step

			np = Vector3.new(np.X, highArena and settleY(r) or gy, np.Z)
			r.CFrame = CFrame.new(np, Vector3.new(pos.X, np.Y, pos.Z))
			r.AssemblyLinearVelocity = Vector3.zero
		end
		return false
	end

	local function wordState(t)
		t = t:lower()
		if t:find("%f[%a]off%f[%A]") then return false end
		if t:find("%f[%a]on%f[%A]") then return true end
		return nil

	local evState, evHooked = nil, {}
	local evStateAt, lastReequip, swingEvtObj = 0, 0, nil
	local function readSwingState(btn)
		for _, inst in ipairs({ btn, btn.Parent }) do
			if inst then
				for k, v in pairs(inst:GetAttributes()) do
					if type(v) == "boolean" then
						local n = k:lower()
						if n:find("auto") or n:find("enabled") or n:find("active") or n == "on" or n:find("toggle") or n:find("state") then
							return v
						end
					end
				end
			end
		end
		for _, d in ipairs(btn:GetChildren()) do
			if d:IsA("BoolValue") then return d.Value end
		end
		local nodes = btn:GetDescendants()
		table.insert(nodes, 1, btn)
		for _, d in ipairs(nodes) do
			if d:IsA("TextLabel") or d:IsA("TextButton") then
				local r = wordState(d.Text)
				if r ~= nil then return r end
			end
		end

		if evState ~= nil and evStateAt >= lastReequip then return evState end
		return nil
	end

	local function hookSwingEvents()
		if type(getnilinstances) ~= "function" then return end
		local ok, list = pcall(getnilinstances)
		if not ok or type(list) ~= "table" then return end
		for _, o in ipairs(list) do
			local isEvt = false
			pcall(function() isEvt = o.Name == "Event" and o:IsA("BindableEvent") end)
			if isEvt and not evHooked[o] then
				evHooked[o] = true
				pcall(function()
					track(o.Event:Connect(function(v)
						if type(v) == "boolean" then
							evState = v
							evStateAt = os.clock()
							if os.clock() - lastSelfPress < 1.5 then swingEvtObj = o end
						end
					end))
				end)
			end
		end
	end
	local function setSwingEvent(want)
		if type(getnilinstances) ~= "function" then return false end
		local ok, list = pcall(getnilinstances)
		if not ok or type(list) ~= "table" then return false end
		for _, obj in ipairs(list) do
			local matches = false
			pcall(function()
				matches = obj.Name == "Event" and obj:IsA("BindableEvent") and obj:GetDebugId() == "0_132375"
			end)
			if matches and pcall(function() obj:Fire(want) end) then
				evState = want
				evStateAt = os.clock()
				return true
			end
		end
		if swingEvtObj and pcall(function() swingEvtObj:Fire(want) end) then
			evState = want
			evStateAt = os.clock()
			return true
		end
		return false
	end

	local swingAssumed = false
	local swingBeforeDoor = nil
	local function captureSwing()
		local b = findSwingBtn()
		local st = b and readSwingState(b)
		if st == nil then st = evState end
		swingBeforeDoor = st
	end
	local function ensureSwingBtn()
		local btn = findSwingBtn()
		for _ = 1, 3 do
			if btnShown(btn) then return btn end
			reequipPickaxe()
			task.wait(0.6)
			btn = findSwingBtn()

	local function enableAutoSwingCore()
		hookSwingEvents()
		swingAssumed = false
		local btn = ensureSwingBtn()
		if not btn then
			if setSwingEvent(true) then swingAssumed = true end
			status("AutoSwing button not found - using event only")
			return
		end
		local pressed = false
		local st = readSwingState(btn)
		if st == false then
			pressed = pressBtn(btn)
			task.wait(0.5)
		elseif st == nil then
			-- unknown state: ask through the event first, a blind press could turn an ON toggle OFF
			local evOk = setSwingEvent(true)
			task.wait(0.4)
			st = readSwingState(btn)
			if st == false or (st == nil and not evOk and swingBeforeDoor ~= true) then
				pressed = pressBtn(btn)
				task.wait(0.5)
			end
		end
		local fired = setSwingEvent(true)
		if pressed or fired or readSwingState(btn) == true or swingBeforeDoor == true then swingAssumed = true end

		for _ = 1, 3 do
			if readSwingState(btn) ~= false then break end
			pressBtn(btn)
			task.wait(0.5)
		end
	end
	local function enableAutoSwing()
		EV.Enabling = true
		pcall(enableAutoSwingCore)
		EV.Enabling = false
	end
	local function swingText()
		local btn = findSwingBtn()

		if st == nil then st = evState end
		return st == true and "ON" or (st == false and "OFF" or "?")
	end

	local function monitorSwing(gen)
		local lastPress, lastHook, lastEquip, lastForce = 0, 0, 0, 0
		local hookedBtn, userTouched = nil, 0
		while alive(gen) and EV.Mon == gen do
			task.wait(0.15)
			while EV.Enabling and alive(gen) and EV.Mon == gen do task.wait(0.1) end

			if EV.Phase == "door" then
				-- outside the arena nothing to manage; once inside, keep AutoSwing alive
				local okA, inA = false, false
				if EventGate.InArena then okA, inA = pcall(EventGate.InArena) end
				if not (okA and inA == true) then continue end
			end
			local needPick = EV.Phase ~= "boss"

			do
				local b = findSwingBtn()
				if b and b ~= hookedBtn and b:IsA("GuiButton") then
					hookedBtn = b
					local function onClick()
						if os.clock() - lastSelfPress > 0.3 then userTouched = os.clock() end
					end
					pcall(function() track(b.MouseButton1Click:Connect(onClick)) end)
					pcall(function() track(b.Activated:Connect(onClick)) end)
				end
			end
			if userTouched > 0 and os.clock() - userTouched > 0.4 then
				userTouched = 0
				local b = findSwingBtn()
				if b and readSwingState(b) == false then pressBtn(b) end
				setSwingEvent(true)
				swingAssumed = true
			end

			do
				local b = findSwingBtn()
				if os.clock() - lastForce > 1.5 and (not b or readSwingState(b) ~= true) and (pickaxeEquipped() or not needPick) then
					lastForce = os.clock()
					setSwingEvent(true)
				end
			end
			if os.clock() - lastHook > 5 then
				lastHook = os.clock()
				hookSwingEvents()
			end
			if needPick and not pickaxeEquipped() and os.clock() - lastEquip > 1 then
				lastEquip = os.clock()
				equipPickaxe()
			end
			if evState == false then setSwingEvent(true) end
			local btn = findSwingBtn()
			if needPick and not btnShown(btn) and pickaxeEquipped() then btn = ensureSwingBtn() end
			if btn and readSwingState(btn) == false and os.clock() - lastPress > 0.6 and os.clock() - lastSelfPress > 0.7 then
				lastPress = os.clock()
				pressBtn(btn)
				swingAssumed = true
			end
		end
	end
	local function stopAutoSwing()
		EV.Mon = nil
		if not swingAssumed then return end
		if swingBeforeDoor == true then
			swingAssumed = false
			swingBeforeDoor = nil
			return
		end
		equipPickaxe()
		task.wait(0.3)
		local btn = findSwingBtn()
		if btn and btnShown(btn) then
			local cur = readSwingState(btn)
			if cur == nil then cur = evState end
			if cur == nil then cur = swingAssumed end
			if cur then pressBtn(btn) end
		end
		setSwingEvent(false)
		swingAssumed = false
	end

	local function inArena()
		local r = getRoot()
		if not r then return false end
		if r.Position.Y > 500 then return true end
		local a = arena()
		local sp = a and a:FindFirstChild("EggsSpawn")
		return sp ~= nil and sp:IsA("BasePart") and (sp.Position - r.Position).Magnitude < 700
	end

	EventGate.InArena = inArena

	local function fmtTime(sec)
		sec = math.max(0, math.floor(sec))
		return ("%d:%02d"):format(math.floor(sec / 60), sec % 60)
	end

	local function stepDoor(gen)
		captureSwing()
		if inArena() then
			EventGate.Active = true
			return true
		end

		while alive(gen) do
			local joinable = Workspace:GetAttribute("HellJoinable")
			if not joinable then
				EV.WaitClose = false
			else
				local id = Workspace:GetAttribute("HellStartedAt")
				if id ~= nil and id ~= EV.LastId then EV.WaitClose = false end
			end
			if joinable and not EV.WaitClose then break end
			if inArena() then return true end
			local nextAt = Workspace:GetAttribute("HellNextAt")
			local cd = nextAt and (nextAt - os.time()) or 0
			status((joinable and "1/3 Door: waiting for next event" or "1/3 Door: waiting for event")
				.. (cd > 0 and (" (starts in " .. fmtTime(cd) .. ")") or ""))
			task.wait(0.5)
		end
		if not alive(gen) then return false end

		EventGate.Active = true
		EventGate.Abort = false
		local gt = os.clock()
		while alive(gen) and EventGate.StealBusy and os.clock() - gt < 20 do
			status("1/3 Door: waiting for Auto Steal to finish its target")
			task.wait(0.2)
		end
		EventGate.Abort = true
		if not alive(gen) then return false end

		local t0 = os.clock()

		local r = getRoot()
		local SAFE = EventGate.Safe or Vector3.new(-72.5, 3.693, 0)
		local inZone = false
		if r and EventGate.InZone then
			local ok, v = pcall(EventGate.InZone, r.Position)
			inZone = ok and v == true
		end

		local nearSafe = r and (Vector3.new(r.Position.X - SAFE.X, 0, r.Position.Z - SAFE.Z)).Magnitude < 8
		if r and not inZone and not nearSafe and not EV.SafeDone then
			status("1/3 Door: walking to safe zone")
			walkTo(Vector3.new(SAFE.X, r.Position.Y, SAFE.Z), gen, 6, function() return not inArena() end)
		else
			status(EV.SafeDone and "1/3 Door: safe zone already done, going to portal"
				or "1/3 Door: already in a zone, skipping safe zone")
		end
		if alive(gen) then EV.SafeDone = true end

		t0 = os.clock()
		local triedSafe = false
		while alive(gen) and os.clock() - t0 < EV.Timeout.Door do
			if inArena() then return true end
			local portal = Workspace:FindFirstChild("JoinHell", true)
			local tp = portal and (portal:FindFirstChild("TP", true) or portal)
			if not tp and not triedSafe then

				triedSafe = true
				local r2 = getRoot()
				if r2 then
					status("1/3 Door: portal not loaded -> walking to safe zone")
					walkTo(Vector3.new(SAFE.X, r2.Position.Y, SAFE.Z), gen, 6, function() return not inArena() end)
				end
			elseif tp and Workspace:GetAttribute("HellJoinable") then
				status("1/3 Door: walking to portal")
				if walkTo(posOf(tp), gen, 4, function() return not inArena() end) then
					status("1/3 Door: entering portal")
					captureSwing()
					fire("HellJoinRemote")
					if firetouchinterest and tp:IsA("BasePart") and getRoot() then
						pcall(function()
							firetouchinterest(getRoot(), tp, 0)
							firetouchinterest(getRoot(), tp, 1)
						end)
					end
					if fireproximityprompt then
						for _, pp in ipairs(portal:GetDescendants()) do
							if pp:IsA("ProximityPrompt") then pcall(fireproximityprompt, pp) end
						end
					end

					for _ = 1, 40 do
						if not alive(gen) then return false end
						if inArena() then return true end
						task.wait(0.2)
					end
				end
			else
				task.wait(0.5)
			end
			task.wait(0.5)
		end
		return inArena()
	end

	local function isLiveEgg(d)
		if d.Name ~= "Egg" or not d:IsA("BasePart") then return false end
		local hp = d:GetAttribute("Health")
		return hp ~= nil and hp > 0 and not d:GetAttribute("Broken")
	end
	local lastWide = 0
	local function nearestEgg()
		local r = getRoot()
		if not r then return end
		local function scan(container, needEssence)
			local best, bd = nil, math.huge
			for _, d in ipairs(container:GetDescendants()) do
				if isLiveEgg(d) and (not needEssence or d:GetAttribute("EssenceValue") ~= nil) then
					local dist = (d.Position - r.Position).Magnitude
					if dist < bd then best, bd = d, dist end
				end
			end
			return best
		end
		local a = arena()
		local folder = a and (a:FindFirstChild("EssenceEggs") or a)
		local egg = folder and scan(folder, false)
		if egg then return egg end

		if os.clock() - lastWide > 2 then
			lastWide = os.clock()
			return scan(Workspace, true)
		end
	end

	local function eggPhaseOver()
		local n, goal = points()
		if n >= goal then return true end
		if Workspace:GetAttribute("HellGathering") == false then return true end
		local boss = getBoss()
		return boss ~= nil and boss:GetAttribute("Phase") ~= nil and boss:GetAttribute("Phase") ~= "Gather"
	end

	local function eggStatus(extra)
		local n, goal = points()
		status(("2/3 Eggs: %d / %d | Pickaxe %s | AutoSwing %s%s"):format(
			n, goal, pickaxeEquipped() and "OK" or "NOT EQUIPPED", swingText(), extra or ""))
	end

	local EGG_STOP = 6

	local function stepEggsLoop(gen)
		local t0 = os.clock()
		while alive(gen) and os.clock() - t0 < EV.Timeout.Eggs do
			if eggPhaseOver() then return true end
			local egg = nearestEgg()
			if not egg then
				eggStatus(" | searching eggs...")
				task.wait(0.5)
			else
				eggStatus(" | walking to egg")
				local function eggOk() return egg.Parent and egg:GetAttribute("Broken") ~= true and (egg:GetAttribute("Health") or 1) > 0 end
				if walkTo(egg.Position, gen, EGG_STOP, eggOk) then
					local e0, lastS = os.clock(), 0
					while alive(gen) and eggOk() and os.clock() - e0 < 10 do
						local r = getRoot()
						if r and Vector3.new(egg.Position.X - r.Position.X, 0, egg.Position.Z - r.Position.Z).Magnitude > EGG_STOP * 4 then break end

						if r and r.Position.Y > 500 then
							local ty = settleY(r)
							if ty < r.Position.Y - 0.1 then
								r.CFrame = CFrame.new(r.Position.X, ty, r.Position.Z) * (r.CFrame - r.CFrame.Position)
								r.AssemblyLinearVelocity = Vector3.zero
							end
						end

						task.wait(0.15)
						if os.clock() - lastS > 1 then
							lastS = os.clock()
							eggStatus(" | breaking egg")
						end
					end
				end
			end
		end
		return false
	end

	local function stepEggs(gen)
		status("2/3 Eggs: checking pickaxe...")
		hold(false)
		EV.Phase = "eggs"
		ensurePickaxe(gen)
		enableAutoSwing()
		for _ = 1, 4 do
			if not alive(gen) then break end
			task.wait(0.6)
			if swingText() ~= "OFF" then break end
			status("2/3 Eggs: AutoSwing still OFF -> enabling again")
			equipPickaxe()
			enableAutoSwing()
		end
		status(("2/3 Eggs: AutoSwing %s (before door: %s)"):format(swingText(),
			swingBeforeDoor == true and "ON" or (swingBeforeDoor == false and "OFF" or "?")))
		task.wait(1)
		return stepEggsLoop(gen)
	end

	local function stepBoss(gen)
		local t0, seen = os.clock(), false
		EV.Phase = "boss"
		hold(true)
		while alive(gen) and os.clock() - t0 < EV.Timeout.Boss do
			local boss = getBoss()
			local hp = boss and boss:GetAttribute("Health")
			if boss and hp and hp > 0 then
				seen = true
				status(("3/3 Boss: %d / %d"):format(hp, boss:GetAttribute("MaxHealth") or hp))
				tpTo(boss)

				findTool("Bat")
				task.wait(0.15)
			elseif seen or (boss and hp and hp <= 0) then
				return true
			else
				task.wait(0.5)
			end
		end
		return false
	end

	local evToggle
	local function releaseGate()
		EventGate.Active = false
		EventGate.Abort = false
	end

	local function evStop()
		EV.Running = false
		EventGate.Running = false
		EV.Gen += 1
		restoreEventNoclip()
		releaseGate()
		hold(false)
		task.spawn(stopAutoSwing)
	end
	local function evStart()
		if EV.Running then return true end
		EV.Running = true
		EV.Gen += 1
		local gen = EV.Gen
		EventGate.Running = true
		EV.Phase = "door"
		EV.Mon = gen

		task.spawn(monitorSwing, gen)
		task.spawn(function()

			while alive(gen) do
				EV.Phase = "door"
				local entered = stepDoor(gen)
				local ok = entered
				if ok then ok = stepEggs(gen) elseif alive(gen) then status("Stopped: could not enter door") end
				if ok then ok = stepBoss(gen) elseif entered and alive(gen) then status("Stopped: egg goal not reached") end
				if not alive(gen) then break end
				if ok then
					status("Done - event finished")
					Notify("Hell Event finished", true)
				end
				if entered then
					EV.WaitClose = true
					EV.LastId = Workspace:GetAttribute("HellStartedAt")
				end
				releaseGate()
				hold(false)
				EV.Phase = "door"

				local tl = os.clock()
				while alive(gen) and entered and inArena() and os.clock() - tl < 180 do
					status("Event finished - waiting to leave arena")
					task.wait(0.5)
				end
				task.wait(entered and 1 or 3)
			end
			if EV.Gen == gen then
				releaseGate()
				hold(false)
				stopAutoSwing()
				EV.Running = false
				EventGate.Running = false
				restoreEventNoclip()
			end
		end)
		return true
	end

	Section(Page, "Event Control")
	setStatus = StatusCard(Page, "Stopped")
	local selectedEvent = "Blood Moon / Halloween"
	local monitorOn, monitorJob = false, 0

	local function countdownText(timestamp)
		if type(timestamp) ~= "number" then return nil end
		local remaining = math.floor(timestamp - os.time())
		if remaining <= 0 then return "due / time passed" end
		return fmtTime(remaining)
	end

	local function observedEventStatus()
		if selectedEvent == "Blood Moon / Halloween" then
			local blood = Workspace:GetAttribute("BloodMoonActive") == true
			local halloween = Workspace:GetAttribute("HalloweenEnvironmentActive") == true
			if blood then return "Blood Moon: ACTIVE" end
			if halloween then return "Halloween environment: ACTIVE" end
			local raw = Workspace:GetAttribute("HalloweenScheduleState")
			if type(raw) == "string" then
				local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
				if ok and type(data) == "table" then
					local key = tostring(data.NextKey or "unknown")
					local cd = countdownText(tonumber(data.NextAt))
					return "Inactive | next: " .. key .. (cd and (" | " .. cd) or "")
				end
			end
			return "No active state / schedule available"
		elseif selectedEvent == "Halloween Candy Heist" then
			local remote = ReplicatedStorage:FindFirstChild("HalloweenCandyHeist")
			local drops = ReplicatedStorage:FindFirstChild("HalloweenCandyDrops")
			local doors = ReplicatedStorage:FindFirstChild("HalloweenDoors")
			if Workspace:FindFirstChild("CandyHeist", true) then
				return "Candy Heist object found; active state not confirmed"
			end
			if remote or drops or doors then
				return "Event objects found; active state not exposed in Dump"
			end
			return "Candy Heist objects not found in current session"
		elseif selectedEvent == "Titanic Egg" then
			local nextAt = Workspace:GetAttribute("TitanicNextAt")
			local spawnId = Workspace:GetAttribute("TitanicSpawnId")
			local cd = countdownText(tonumber(nextAt))
			return "Spawn ID: " .. tostring(spawnId or "unknown") .. (cd and (" | next: " .. cd) or " | next time unavailable")
		elseif selectedEvent == "Hell Event (existing farm)" then
			if EV.Running then return lastStatus end
			if Workspace:GetAttribute("HellJoinable") == true then return "Hell Event joinable" end
			local cd = countdownText(tonumber(Workspace:GetAttribute("HellNextAt")))
			return "Waiting for Hell Event" .. (cd and (" | " .. cd) or "")
		end
		return "Unknown event"
	end

	local function startMonitor()
		if monitorOn then return end
		monitorOn = true
		monitorJob += 1
		local job = monitorJob
		if selectedEvent == "Hell Event (existing farm)" then
			status("Started existing Hell Event farm")
			evStart()
		else
			status(observedEventStatus())
			Notify("Monitoring only: this Dump does not document a verified farm sequence for this event", false)
		end
		task.spawn(function()
			while monitorOn and monitorJob == job do
				if selectedEvent ~= "Hell Event (existing farm)" then
					status(observedEventStatus())
				end
				task.wait(1)
			end
		end)
	end

	local function stopMonitor()
		monitorOn = false
		monitorJob += 1
		if EV.Running then evStop() end
		status("Stopped")
	end

	Dropdown(Page, "Event Selection", {
		Options = { "Blood Moon / Halloween", "Halloween Candy Heist", "Titanic Egg", "Hell Event (existing farm)" },
		Default = selectedEvent,
		Placeholder = selectedEvent,
		Desc = "Choose an event found in the Dump. Only the existing Hell Event has a verified farm sequence in this source.",
		OnChange = function(v)
			if type(v) ~= "string" or v == "" then return end
			if selectedEvent == "Hell Event (existing farm)" and EV.Running then evStop() end
			selectedEvent = v
			if monitorOn then
				if selectedEvent == "Hell Event (existing farm)" then
					status("Starting existing Hell Event farm")
					evStart()
				else
					if EV.Running then evStop() end
					status(observedEventStatus())
					Notify("Monitoring only: farm sequence is not verified for this event", false)
				end
			else
				status(observedEventStatus())
			end
		end,
	})

	local _, monitorToggle = Toggle(Page, "Start Event Monitor", false, function(on)
		if on then startMonitor() else stopMonitor() end
	end, nil, "Manual start/stop. New Dump events are status/countdown only until their farm steps are confirmed.")

	addCleanup(function()
		monitorOn = false
		monitorJob += 1
		evStop()
	end)
	end,
}

local Lunar = fetchLibrary()
if not Lunar then
	warn("[NEXOEA Loader] Could not load NEXOEA Library. Set CONFIG.LibraryUrl or put NEXOEA_Library.lua in your workspace folder.")
	return
end

local systems = {}
for _, system in ipairs(ALL_SYSTEMS) do
	if CONFIG.Systems[system.Id] ~= false then
		table.insert(systems, system)
	end
end

local Hub = Lunar.new({
	Id = CONFIG.Id or tostring(game.PlaceId),
	GameName = CONFIG.GameName,
	Translations = TRANSLATIONS,
	ConfigExclude = { "^Farm:AFK", "^Section:Farm:AFK" },
	Systems = systems,
})

if not Hub then
	return
end

local Loader = {
	Hub = Hub,
	Config = CONFIG,
	Systems = systems,
	EventGate = EventGate,
	Unload = function()
		pcall(Hub.Destroy)
		ENV.LunarLoader = nil
	end,
}
ENV.LunarLoader = Loader

log(("ready - %d system(s): %s"):format(#systems, table.concat((function()
	local names = {}
	for _, s in ipairs(systems) do
		table.insert(names, s.Name)
	end
	return names
end)(), ", ")))

return Loader
